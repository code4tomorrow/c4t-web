You won't start from a blank sketch. The starter below already drives two motors through driver signals, reads a filtered distance, stops on a button press, and logs everything, because you built each of those pieces in modules 5 to 7. What it doesn't have is *behavior*. Three states are marked `TODO`, and filling them in is the capstone.

## What we're building

The scanning avoider, the intermediate project from the last lesson. The other projects reuse most of this; the line-follower changes come at the end.


## The starter sketch

Paste this into `sketch.ino`. Read it top to bottom once before changing anything; every function in it comes from an earlier module.

```arduino
#include <Servo.h>

// ---------- pins (from the pin plan) ----------
const int ENA = 5, IN1 = 7, IN2 = 8;     // left motor driver inputs
const int ENB = 6, IN3 = 4, IN4 = 3;     // right motor driver inputs
const int SCAN_SERVO = 9;
const int TRIG = 10, ECHO = 11;
const int BUTTON = 2;                    // start/stop, wired to GND

// ---------- numbers you will tune ----------
const int CRUISE = 160;                  // forward speed, 0-255
const int MAX_SPEED = 200;               // drive() never goes faster (safety S3)
const long STOP_CM = 20;                 // react when closer than this
const unsigned long SAMPLE_MS = 60;      // time between sensor pings
const unsigned long LOG_MS = 200;        // time between log lines
const bool LOGGING = true;

enum State { WAITING, DRIVING, BACKING_UP, SCANNING, TURNING };
const char* NAMES[] = {"WAITING", "DRIVING", "BACKING_UP", "SCANNING", "TURNING"};
State state = WAITING;
unsigned long stateStart = 0;

Servo scanner;
long history[5] = {-1, -1, -1, -1, -1};
int nextSlot = 0;
long distanceCm = -1;                    // filtered; -1 means no valid reading
unsigned long lastSample = 0, lastLog = 0;
int leftCmd = 0, rightCmd = 0;           // last motor commands, for the log
int lastRaw = HIGH, stableButton = HIGH;
unsigned long lastChange = 0;

// ---------- motors ----------
void setChannel(int en, int a, int b, int speed) {
  speed = constrain(speed, -MAX_SPEED, MAX_SPEED);
  digitalWrite(a, speed > 0 ? HIGH : LOW);
  digitalWrite(b, speed < 0 ? HIGH : LOW);
  analogWrite(en, abs(speed));
}

void drive(int left, int right) {
  leftCmd = left;
  rightCmd = right;
  setChannel(ENA, IN1, IN2, left);
  setChannel(ENB, IN3, IN4, right);
}

// ---------- sensor ----------
long readRawCm() {
  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG, LOW);
  unsigned long duration = pulseIn(ECHO, HIGH, 30000UL);
  if (duration == 0) return -1;
  return (duration * 343UL + 10000UL) / 20000UL;
}

long medianCm() {
  long sorted[5];
  int n = 0;
  for (int i = 0; i < 5; i++) {
    if (history[i] >= 0) sorted[n++] = history[i];
  }
  if (n == 0) return -1;
  for (int i = 1; i < n; i++) {
    long value = sorted[i];
    int j = i - 1;
    while (j >= 0 && sorted[j] > value) {
      sorted[j + 1] = sorted[j];
      j--;
    }
    sorted[j + 1] = value;
  }
  return sorted[n / 2];
}

void sampleSensor() {
  history[nextSlot] = readRawCm();
  nextSlot = (nextSlot + 1) % 5;
  distanceCm = medianCm();
}

// ---------- button: true once per press, after bounce settles ----------
bool buttonPressed() {
  int raw = digitalRead(BUTTON);
  if (raw != lastRaw) {
    lastRaw = raw;
    lastChange = millis();
  }
  if (millis() - lastChange >= 30 && raw != stableButton) {
    stableButton = raw;
    return stableButton == LOW;
  }
  return false;
}

// ---------- states and logging ----------
void enterState(State next) {
  Serial.print(millis());
  Serial.print(" ms: ");
  Serial.print(NAMES[state]);
  Serial.print(" -> ");
  Serial.println(NAMES[next]);
  state = next;
  stateStart = millis();
}

void logStatus(unsigned long now) {
  if (!LOGGING || now - lastLog < LOG_MS) return;
  lastLog = now;
  Serial.print(now);         Serial.print(',');
  Serial.print(NAMES[state]); Serial.print(',');
  Serial.print(distanceCm);  Serial.print(',');
  Serial.print(leftCmd);     Serial.print(',');
  Serial.println(rightCmd);
}

void setup() {
  int outputs[] = {ENA, IN1, IN2, ENB, IN3, IN4, TRIG};
  for (int i = 0; i < 7; i++) pinMode(outputs[i], OUTPUT);
  drive(0, 0);                           // safety S1: motors off before anything else
  pinMode(ECHO, INPUT);
  pinMode(BUTTON, INPUT_PULLUP);
  scanner.attach(SCAN_SERVO);
  scanner.write(90);                     // look straight ahead
  Serial.begin(115200);
  Serial.println("ms,state,cm,left,right");
}

void loop() {
  unsigned long now = millis();
  if (now - lastSample >= SAMPLE_MS) {
    lastSample = now;
    sampleSensor();
  }
  unsigned long inState = now - stateStart;   // the TODO states will need this
  bool pressed = buttonPressed();

  if (pressed && state != WAITING) {     // safety S2: a press stops any moving state
    enterState(WAITING);
    pressed = false;                     // this press is used up
  }

  switch (state) {
    case WAITING:
      drive(0, 0);
      scanner.write(90);
      if (pressed) enterState(DRIVING);
      break;

    case DRIVING:
      drive(CRUISE, CRUISE);
      if (distanceCm >= 0 && distanceCm < STOP_CM) enterState(BACKING_UP);
      break;

    case BACKING_UP:
      // TODO milestone 3: reverse for 400 ms, then go to TURNING
      drive(0, 0);
      break;

    case SCANNING:
      // TODO milestone 4: look left, look right, pick a side
      drive(0, 0);
      break;

    case TURNING:
      // TODO milestone 3: spin until the path is clear, with a time limit
      drive(0, 0);
      break;
  }

  logStatus(now);
}
```

Run it, press START/STOP, and drag the HC-SR04 slider below 20 cm. The log shows the robot go from WAITING to DRIVING to BACKING_UP, and then sit there with the motors off, because BACKING_UP is still a `TODO`:

```text
ms,state,cm,left,right
...
1030 ms: WAITING -> DRIVING
4800,DRIVING,32,160,160
5040 ms: DRIVING -> BACKING_UP
5200,BACKING_UP,10,0,0
```

Notice the log keeps running during every state, and the button works in every state. That's the payoff of the non-blocking structure: you can add behavior without breaking either.

## Milestone 1: Wiring and motor self-test

**Goal:** prove every driver signal is on the pin the code thinks it is, before any behavior depends on it. Write `selfTest()` and call it at the end of `setup()`: each wheel forward then backward for 400 ms, one at a time, then the servo to 30°, 90°, and 150°.

**Demo:** the LEDs light in this order: L-EN with L-IN1, L-EN with L-IN2, R-EN with R-IN3, R-EN with R-IN4. Then the servo steps across. On a real robot, do this with the wheels in the air (safety S6).

:::solution
```arduino
void selfTest() {
  int pattern[4][2] = {{150, 0}, {-150, 0}, {0, 150}, {0, -150}};
  for (int i = 0; i < 4; i++) {
    drive(pattern[i][0], pattern[i][1]);
    delay(400);
    drive(0, 0);
    delay(200);
  }
  for (int angle = 30; angle <= 150; angle += 60) {
    scanner.write(angle);
    delay(400);
  }
  scanner.write(90);
}
```

Using `delay()` is fine here. The self-test runs once, before the robot is allowed to drive, and it's meant to be watched. If the wrong LED lights, fix the wiring or the pin constants now; every later milestone assumes this one passed.
:::

## Milestone 2: Sensor, filter, and log

**Goal:** confirm the sensor pipeline works in *this* circuit. The starter already pings every 60 ms, rounds, and filters with a median of five.

**Demo:** in WAITING, set the slider to 30, 100, and 250 cm; the `cm` column should be within a centimeter or so of each. Then jump from 100 to 15 and count log lines: `cm` should follow within about three samples. If you want to test fault handling here too, paste `injectFault()` and the slide switch from the module 7 harness into this sketch.

## Milestone 3: The core states

**Goal:** fill in BACKING_UP and TURNING so the robot backs up for 400 ms, then spins left until the path is clear beyond 35 cm, with at least 400 ms of turning (no twitching) and at most 3 seconds (safety S5).

**Demo:** with the slider below 20 cm, the log shows DRIVING, BACKING_UP, TURNING. Raise the slider above 35 cm during the turn and it returns to DRIVING. Leave the slider low and after 3 seconds it gives up and goes to WAITING.

:::hint
BACKING_UP only needs a drive command and one timed exit. TURNING has two exits: the clear-path one (time *and* distance) and the time limit. Decide what a reading of -1 (no echo at all) should mean when you're checking for a clear path.
:::

:::solution
```arduino
    case BACKING_UP:
      drive(-140, -140);
      if (inState >= 400) enterState(TURNING);
      break;

    case TURNING:
      drive(-140, 140);                                   // spin left
      if (inState >= 400 && (distanceCm < 0 || distanceCm > 35)) enterState(DRIVING);
      else if (inState >= 3000) enterState(WAITING);      // give up; a human decides
      break;
```

A filtered -1 means no echo for five pings in a row, which in open space means "nothing in range", so it counts as clear here. The minimum 400 ms is the hysteresis in time; the 35 cm exit versus the 20 cm entry is the hysteresis in distance.
:::

## Milestone 4: Look before you turn

**Goal:** after backing up, stop, point the servo left and take a reading, point it right and take another, face forward, then turn toward the side with more room.

There's a Wokwi catch. The simulated sensor doesn't know where the servo points; the slider sets one distance for every direction. So *you* play the room: while the servo looks left, set the slider to the left-hand distance, then change it while the servo looks right. To give your hand time, each look lasts `LOOK_MS`. Use 2000 ms in Wokwi; about 500 ms is plenty on a real robot (enough for the servo to arrive and the median to refill).

Write the decision rule in Python first and test the awkward cases, including a side with no echo and a tie:

```python run
def as_range(cm):
    return 400 if cm < 0 else cm          # no echo counts as far away

def choose_turn(left_cm, right_cm):
    return "left" if as_range(left_cm) > as_range(right_cm) else "right"

cases = [
    (120, 40, "left"),
    (30, 90, "right"),
    (-1, 80, "left"),     # nothing echoed on the left: open space
    (50, 50, "right"),    # a tie: any fixed rule is fine, as long as it's fixed
    (15, 18, "right"),    # both blocked: turn anyway; TURNING's time limit catches it
]
for left, right, expected in cases:
    got = choose_turn(left, right)
    print(f"left={left:4} right={right:4} -> {got:5} {'PASS' if got == expected else 'FAIL'}")
```

:::hint
SCANNING has three phases: look left, look right, face forward. Keep the phase in a global `int scanPhase`, reset it to 0 when BACKING_UP hands over, and record each reading at the end of its phase, when the filter has had time to refill.
:::

:::solution
New globals near the top:

```arduino
const unsigned long LOOK_MS = 2000;   // 2 s to move the slider in Wokwi; about 500 on a real robot
long leftCm = -1, rightCm = -1;
int turnDir = 1;                      // +1 spins right, -1 spins left
int scanPhase = 0;

long asRange(long cm) {
  return cm < 0 ? 400 : cm;
}
```

The three states that change:

```arduino
    case BACKING_UP:
      drive(-140, -140);
      if (inState >= 400) {
        scanPhase = 0;
        enterState(SCANNING);
      }
      break;

    case SCANNING:
      drive(0, 0);
      if (scanPhase == 0) {
        scanner.write(150);                         // look left
        if (inState >= LOOK_MS) {
          leftCm = asRange(distanceCm);
          scanPhase = 1;
        }
      } else if (scanPhase == 1) {
        scanner.write(30);                          // look right
        if (inState >= 2 * LOOK_MS) {
          rightCm = asRange(distanceCm);
          scanPhase = 2;
        }
      } else {
        scanner.write(90);                          // face forward again
        if (inState >= 2 * LOOK_MS + 300) {
          turnDir = (leftCm > rightCm) ? -1 : 1;
          Serial.print("scan: left ");               // evidence for your report
          Serial.print(leftCm);
          Serial.print(" cm, right ");
          Serial.println(rightCm);
          enterState(TURNING);
        }
      }
      break;

    case TURNING:
      drive(140 * turnDir, -140 * turnDir);
      if (inState >= 400 && (distanceCm < 0 || distanceCm > 35)) enterState(DRIVING);
      else if (inState >= 3000) enterState(WAITING);
      break;
```

The servo faces forward before TURNING starts, so the turn ends on what's *ahead*, not on what was to the side. If your servo is mounted the other way round, the robot will turn toward the closer side; swap 150 and 30.
:::

## Milestone 5: Safety features

**Goal:** finish the safety checklist. The starter already covers S1 (motors off first), S2 (button stops anything), and S3 (`MAX_SPEED`). Add:

1. **A start delay.** After START, wait one second before the wheels turn, so a hand can get clear.
2. **A stuck watchdog** in DRIVING, from module 7: if the distance hasn't changed by 2 cm in 3 seconds while something is within 150 cm, log `STUCK` and go to WAITING.

:::solution
Two globals, set every time DRIVING begins (add the `if` to the end of `enterState()`):

```arduino
long refCm = -1;
unsigned long refTime = 0;

// at the end of enterState():
if (next == DRIVING) {
  refCm = distanceCm;
  refTime = millis();
}
```

The new DRIVING case:

```arduino
    case DRIVING:
      if (inState < 1000) drive(0, 0);              // start delay
      else drive(CRUISE, CRUISE);
      if (distanceCm >= 0 && distanceCm < STOP_CM) {
        enterState(BACKING_UP);
      } else if (labs(distanceCm - refCm) >= 2) {
        refCm = distanceCm;                         // it moved: restart the watchdog
        refTime = now;
      } else if (now - refTime >= 3000 && distanceCm >= 0 && distanceCm < 150) {
        Serial.println("STUCK: distance unchanged for 3 s");
        enterState(WAITING);
      }
      break;
```

In Wokwi the robot never really moves, so if you leave the slider under 150 cm during DRIVING, the watchdog fires after 3 seconds. That isn't a false alarm; it's the watchdog correctly noticing a robot that isn't getting anywhere. Drag the slider slowly toward the robot to simulate driving, and it stays quiet.
:::

## Variant: the line follower

For a line-follower project, swap the HC-SR04 and servo for two **photoresistor sensor** modules: AO of the left one to A0, the right one to A1, VCC to 5V, GND to GND. Each module has a light slider; with this module, less light means a *higher* reading, so set a sensor to about 10 lux for "over the dark line" and leave it at 500 lux for "white floor".

The state list becomes WAITING, CALIBRATING, FOLLOWING, SEARCHING. Calibration and following look like this; SEARCHING is the one from the module 6 challenge:

```arduino
const int LEFT_SENSOR = A0, RIGHT_SENSOR = A1;
const float KP = 0.15;            // error now spans -1000..1000, so the gain is smaller
int lowL = 1023, highL = 0, lowR = 1023, highR = 0;
unsigned long lastSeen = 0;
int lastError = 0;

int normalized(int pin, int low, int high) {
  if (high - low < 50) return 0;  // never saw the line during calibration
  return map(constrain(analogRead(pin), low, high), low, high, 0, 1000);
}

// inside the switch:
    case CALIBRATING: {           // move both light sliders between floor and line
      drive(0, 0);
      int l = analogRead(LEFT_SENSOR);
      int r = analogRead(RIGHT_SENSOR);
      lowL = min(lowL, l);
      highL = max(highL, l);
      lowR = min(lowR, r);
      highR = max(highR, r);
      if (inState >= 3000) enterState(FOLLOWING);
      break;
    }

    case FOLLOWING: {
      int left = normalized(LEFT_SENSOR, lowL, highL);
      int right = normalized(RIGHT_SENSOR, lowR, highR);
      int error = right - left;     // positive: the line is to the right
      int correction = KP * error;
      drive(CRUISE + correction, CRUISE - correction);
      if (left > 500 || right > 500) {
        lastSeen = now;
        lastError = error;
      } else if (now - lastSeen >= 200) {
        enterState(SEARCHING);
      }
      break;
    }
```

The braces after `case CALIBRATING:` and `case FOLLOWING:` are required because those cases declare variables. In Wokwi you can't drive along a real line, so test the controller by setting the sliders: left dark and right light should push the right wheel faster than the left (the robot steers left, toward the line), and both light for 200 ms should start a search. For the advanced project, add the HC-SR04 back on pins 10 and 11 and a PAUSED state that FOLLOWING enters when something is within 20 cm.

## Moving to a real chassis

The sketch runs unchanged; only the wiring and a few numbers move:

- Wire pins 5, 7, 8, 6, 4, 3 to ENA, IN1, IN2, ENB, IN3, IN4 on the driver (remove the ENA and ENB jumpers on an L298N). Motor pack to the driver's motor supply, grounds joined.
- Mount the HC-SR04 level, about 10 cm above the floor, clear of the wheels. Set `LOOK_MS` to about 500.
- Run milestone 1 with the wheels in the air, then lower `CRUISE` until the robot can stop within `STOP_CM` at full speed. The filter delay from module 7 means a fast robot needs a larger stop distance.
- Expect to retune every number. That's what the next lesson's test plan is for.

:::takeaways
- Start from a skeleton whose infrastructure works, and add behavior one milestone at a time.
- Each milestone ends with a demo you can check in the log.
- Test decision rules in Python, including the awkward cases, before writing them in C++.
- In Wokwi you play the world: the slider is every wall, and it only moves when you do.
- Safety features (start delay, button stop, time limits, watchdog) are part of the build, not extras.
:::
