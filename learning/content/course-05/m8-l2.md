## Milestone 1: Rooms built from data

Start with the errors, because every other module raises them. Two families, on purpose: `GameError` is for things the *player* did (the game replies and carries on), and `WorldError` is for a broken map, which is the *designer's* bug and should stop the program at startup.

```python run title="adventure/errors.py"
class GameError(Exception):
    """Something the player tried that the game refuses. Shown as a reply."""


class UnknownCommand(GameError):
    def __init__(self, word: str) -> None:
        super().__init__(f"I don't know how to {word!r}. Type 'help'.")
        self.word = word


class CantDoThat(GameError):
    """A sensible command that doesn't work right now."""


class WorldError(Exception):
    """The map data is broken: a bug in the map, not a player mistake."""
```

The world is plain data, so you can design new maps without touching code. `World.from_dict` turns it into connected `Room` objects; any missing name becomes a `WorldError`.

```python run title="adventure/world.py"
from dataclasses import dataclass


@dataclass(frozen=True)
class Item:
    name: str
    description: str


class Room:
    def __init__(self, name: str, description: str) -> None:
        self.name = name
        self.description = description
        self.exits: dict[str, "Room"] = {}
        self._locks: dict[str, str] = {}  # direction -> name of the key that opens it
        self._items: dict[str, Item] = {}

    def add_item(self, item: Item) -> None:
        self._items[item.name] = item

    def remove_item(self, name: str) -> Item:
        try:
            return self._items.pop(name)
        except KeyError:
            raise CantDoThat(f"There is no {name} here.") from None

    def lock(self, direction: str, key: str) -> None:
        self._locks[direction] = key

    def key_for(self, direction: str) -> str | None:
        return self._locks.get(direction)

    def unlock(self, direction: str) -> None:
        self._locks.pop(direction, None)

    def describe(self) -> str:
        lines = [f"{self.name.title()}: {self.description}"]
        if self._items:
            lines.append("You see: " + ", ".join(sorted(self._items)))
        lines.append("Exits: " + ", ".join(sorted(self.exits)))
        return "\n".join(lines)


class World:
    def __init__(self, rooms: dict[str, Room], start: Room, goal: str | None) -> None:
        self.rooms = rooms
        self.start = start
        self.goal = goal

    @classmethod
    def from_dict(cls, data: dict) -> "World":
        try:
            items = {name: Item(name, text) for name, text in data.get("items", {}).items()}
            rooms = {name: Room(name, spec["description"]) for name, spec in data["rooms"].items()}
            for name, spec in data["rooms"].items():
                for direction, target in spec.get("exits", {}).items():
                    rooms[name].exits[direction] = rooms[target]
                for direction, key in spec.get("locked", {}).items():
                    rooms[name].lock(direction, key)
                for item_name in spec.get("items", []):
                    rooms[name].add_item(items[item_name])
            return cls(rooms, rooms[data["start"]], data.get("goal"))
        except KeyError as err:
            raise WorldError(f"the map mentions {err}, which isn't defined") from err
```

## Milestone 2: A player with an inventory

`Player` is composition at work: it *has* a current room and *has* an inventory, and it borrows behavior from both instead of inheriting from either.

```python run title="adventure/player.py"
class Player:
    def __init__(self, room: Room) -> None:
        self.room = room
        self._inventory: dict[str, Item] = {}

    def move(self, direction: str) -> None:
        if direction not in self.room.exits:
            raise CantDoThat(f"You can't go {direction} from here.")
        if self.room.key_for(direction):
            raise CantDoThat(f"The way {direction} is locked.")
        self.room = self.room.exits[direction]

    def take(self, name: str) -> Item:
        item = self.room.remove_item(name)
        self._inventory[name] = item
        return item

    def drop(self, name: str) -> None:
        if name not in self._inventory:
            raise CantDoThat(f"You aren't carrying a {name}.")
        self.room.add_item(self._inventory.pop(name))

    def has(self, name: str) -> bool:
        return name in self._inventory

    def inventory(self) -> list[str]:
        return sorted(self._inventory)
```

End the milestone with tests. Each gets a fresh world from `setUp`, and the last two check that refusals raise the right error *and* leave the state unchanged:

```python run title="tests/test_player.py"
import sys
import unittest


def run_tests(*cases: type[unittest.TestCase]) -> None:
    suite = unittest.TestSuite()
    for case in cases:
        suite.addTests(unittest.defaultTestLoader.loadTestsFromTestCase(case))
    unittest.TextTestRunner(stream=sys.stdout, verbosity=2).run(suite)


class TestPlayer(unittest.TestCase):
    def setUp(self):
        self.player = Player(World.from_dict(HOUSE).start)

    def test_moving_changes_room(self):
        self.player.move("west")
        self.assertEqual(self.player.room.name, "kitchen")

    def test_taking_moves_item_from_room_to_inventory(self):
        self.player.take("note")
        self.assertTrue(self.player.has("note"))
        self.assertNotIn("note", self.player.room.describe())

    def test_locked_exit_refuses_and_player_stays(self):
        self.player.move("north")
        with self.assertRaises(CantDoThat):
            self.player.move("east")
        self.assertEqual(self.player.room.name, "corridor")

    def test_taking_a_missing_item_refuses(self):
        with self.assertRaises(CantDoThat):
            self.player.take("sword")
        self.assertEqual(self.player.inventory(), [])


run_tests(TestPlayer)
```

## Milestone 3: Commands as classes

Each command is a class with the same `run(game, args)` method, as in the Module 7 challenge. Before reading the real ones, step through the core idea: a dictionary picks the object, and the same call runs different code.

:::trace One call, many commands
```python
class Look:
    def run(self, args):
        return "You are in the hall."

class Go:
    def run(self, args):
        return f"You walk {args[0]}."

registry = {"look": Look(), "go": Go()}

for line in ["look", "go north"]:
    word, *args = line.split()
    command = registry[word]
    print(command.run(args))
```
:::

```python run title="adventure/commands.py"
from abc import ABC, abstractmethod


class Command(ABC):
    names: tuple[str, ...] = ()
    usage: str = ""
    arity: int = 0  # how many words must follow the command name

    @abstractmethod
    def run(self, game: "Game", args: list[str]) -> str: ...


class Look(Command):
    names, usage = ("look", "l"), "look"

    def run(self, game, args):
        return game.player.room.describe()


class Go(Command):
    names, usage, arity = ("go",), "go DIRECTION", 1

    def run(self, game, args):
        game.player.move(args[0])
        return game.player.room.describe()


class Take(Command):
    names, usage, arity = ("take", "get"), "take ITEM", 1

    def run(self, game, args):
        return f"Taken: {game.player.take(args[0]).name}."


class Drop(Command):
    names, usage, arity = ("drop",), "drop ITEM", 1

    def run(self, game, args):
        game.player.drop(args[0])
        return f"Dropped: {args[0]}."


class Inventory(Command):
    names, usage = ("inventory", "i"), "inventory"

    def run(self, game, args):
        items = game.player.inventory()
        return "You carry: " + (", ".join(items) if items else "nothing")


class Help(Command):
    names, usage = ("help",), "help"

    def run(self, game, args):
        return "Commands: " + "; ".join(command.usage for command in game.commands)


COMMANDS: list[Command] = [Look(), Go(), Take(), Drop(), Inventory(), Help()]
```

`Game` is the boundary. It's the only place that catches `GameError`, and it turns each one into a reply. A `WorldError` or a genuine bug is *not* caught, so it still crashes with a traceback.

```python run title="adventure/game.py"
class Game:
    def __init__(self, world: World, commands: list[Command]) -> None:
        self.world = world
        self.player = Player(world.start)
        self.commands = commands
        self._by_name = {name: command for command in commands for name in command.names}

    @classmethod
    def from_data(cls, data: dict) -> "Game":
        return cls(World.from_dict(data), COMMANDS)

    @property
    def won(self) -> bool:
        return self.player.room.name == self.world.goal

    def handle(self, line: str) -> str:
        words = line.lower().split()
        if not words:
            return "Say something. Type 'help' for commands."
        try:
            command = self._by_name.get(words[0])
            if command is None:
                raise UnknownCommand(words[0])
            if len(words) - 1 != command.arity:
                raise CantDoThat(f"Usage: {command.usage}")
            return command.run(self, words[1:])
        except GameError as err:
            return str(err)
```

`from_data` reads the global `COMMANDS` list each time it's called, so adding a command later means adding one object to that list. Finally the entry point. `read` and `write` default to `input` and `print`, which lets tests drive the whole loop with scripted input later:

```python run title="adventure/__main__.py" input="help|take note|dance|go west|take key|i|go east|go north|go east|quit"
from collections.abc import Callable


def main(data: dict, read: Callable[[str], str] = input, write: Callable[[str], None] = print) -> None:
    game = Game.from_data(data)
    write(game.player.room.describe())
    while not game.won:
        line = read("> ")
        if line.strip().lower() in ("quit", "q"):
            write("Bye!")
            return
        write(game.handle(line))
    write("You reached the vault. You win!")


if __name__ == "__main__":
    main(HOUSE)
```

Every reply in that session came from a different `Command` class, and `dance` hit the `UnknownCommand` branch without stopping the game. There's no way to win yet: the vault is locked and nothing can open it.

## Milestone 4: Locked doors and winning

Your turn first. Write an `Unlock` command (`unlock DIRECTION`) that refuses with `CantDoThat` if nothing is locked that way or the player doesn't have the key, and otherwise calls `room.unlock(direction)`. Add it to `COMMANDS`. You shouldn't need to edit `Game`, `Player`, or any other command. Then rerun the acceptance tests from the last lesson. One line changes: `Game(TINY_WORLD)` becomes `Game.from_data(TINY_WORLD)`, because building the `World` became its own step. Updating *how* a test sets things up as the design improves is normal; changing *what* it checks is not.

```python run title="adventure/commands.py (Milestone 4)"
class Unlock(Command):
    names, usage, arity = ("unlock",), "unlock DIRECTION", 1

    def run(self, game, args):
        direction = args[0]
        room = game.player.room
        key = room.key_for(direction)
        if key is None:
            raise CantDoThat(f"Nothing is locked to the {direction}.")
        if not game.player.has(key):
            raise CantDoThat(f"You need the {key}.")
        room.unlock(direction)
        return f"You unlock the way {direction} with the {key}."


COMMANDS.append(Unlock())

TINY_WORLD = {
    "start": "hall",
    "goal": "vault",
    "items": {"key": "A small brass key."},
    "rooms": {
        "hall": {"description": "A dusty hall.", "exits": {"north": "corridor"}, "items": ["key"]},
        "corridor": {
            "description": "A narrow corridor.",
            "exits": {"south": "hall", "east": "vault"},
            "locked": {"east": "key"},
        },
        "vault": {"description": "Gold, everywhere.", "exits": {"west": "corridor"}},
    },
}


class TestWholeGame(unittest.TestCase):
    def test_locked_door_blocks_the_way(self):
        game = Game.from_data(TINY_WORLD)
        game.handle("go north")
        self.assertIn("locked", game.handle("go east"))

    def test_player_wins_by_unlocking_the_vault(self):
        game = Game.from_data(TINY_WORLD)
        for line in ["take key", "go north", "unlock east", "go east"]:
            game.handle(line)
        self.assertTrue(game.won)

    def test_unlocking_without_the_key_refuses(self):
        game = Game.from_data(TINY_WORLD)
        game.handle("go north")
        self.assertEqual(game.handle("unlock east"), "You need the key.")


run_tests(TestWholeGame, TestPlayer)
```

The acceptance tests you wrote before any code are green. `Unlock` reaches the room only through `key_for` and `unlock`, never through `_locks`, so `Room` stays in charge of its own rules. Now the house can be won; the end of this game is the transcript from the top of the page:

```python run title="play the house" input="go west|take key|go east|go north|go east|unlock east|go east"

```

## Milestone 5: A real package, maps in JSON, and more tests

Now split the cells into files. The imports must flow one way, like the booking package in Module 5:

| File | Imports from the package | Holds |
| --- | --- | --- |
| `adventure/errors.py` | nothing | `GameError`, `UnknownCommand`, `CantDoThat`, `WorldError` |
| `adventure/world.py` | `errors` | `Item`, `Room`, `World` |
| `adventure/player.py` | `errors`, `world` | `Player` |
| `adventure/commands.py` | `errors` | `Command` and every command, `COMMANDS` |
| `adventure/game.py` | `errors`, `world`, `player`, `commands` | `Game` |
| `adventure/__main__.py` | `game`, `maps` | `main()` and the guard |

`commands.py` mentions `Game` only in a type hint, written as the string `"Game"`. Don't import `game.py` there; `game.py` already imports `commands.py`, and the loop would fail with `ImportError: cannot import name ... (most likely due to a circular import)`.

Move the map into `adventure/maps/house.json` so non-programmers can write levels. Loading it is a boundary with the outside world, so translate its failures into `WorldError`:

```python run title="adventure/maps.py (Milestone 5)"
import json
from pathlib import Path


def load_map(path: Path) -> dict:
    try:
        return json.loads(path.read_text())
    except FileNotFoundError as err:
        raise WorldError(f"no map file at {path}") from err
    except json.JSONDecodeError as err:
        raise WorldError(f"{path.name} is not valid JSON (line {err.lineno})") from err


Path("house.json").write_text(json.dumps(HOUSE, indent=2))
print(Game.from_data(load_map(Path("house.json"))).handle("look"))

Path("broken.json").write_text('{"start": "hall",\n "rooms": {')
for path in [Path("broken.json"), Path("missing.json")]:
    try:
        load_map(path)
    except WorldError as err:
        print("WorldError:", err)
```

To finish the milestone, aim for at least 12 tests across `tests/test_world.py`, `tests/test_player.py`, `tests/test_commands.py`, and `tests/test_game.py`. The next lesson has a test plan, the pytest versions, and a way to test `main()` itself with scripted input.

:::warning
**Printing from deep inside.** If `Player.move` printed "You can't go that way" instead of raising, the message would appear in tests, the `Game` couldn't change the wording, and a future web version would print to a server log nobody reads. Classes below `Game` raise or return; only `main()` calls `print`.
:::

## Adapting the starter to your project

- **Library loans:** `Library` plays the role of `Game`; commands like `borrow`, `return`, and `due` become `Command` classes; `DateRange` from Module 6 handles loan periods.
- **Habit tracker:** keep the command registry and the `GameError` boundary (rename it `TrackerError`); put persistence behind a `Storage` ABC with JSON and in-memory versions.
- **Arena:** replace `Command` with a `Move` ABC (`Attack`, `Heal`, `Shield`); `Battle.turn()` plays the role of `Game.handle`.
- **Elevator:** a `Scheduler` ABC with two strategies; the main loop advances a clock instead of reading commands, and tests drive it with fixed request lists.
