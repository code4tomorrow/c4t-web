## Warm-up: whose method runs?

`report` is defined only in `Unit`. Predict the three lines before reading the output.

```python run title="units.py"
class Unit:
    def report(self) -> str:
        return f"{self.label()} deals {self.damage()} damage"

    def label(self) -> str:
        return "unit"

    def damage(self) -> int:
        return 1


class Archer(Unit):
    def label(self) -> str:
        return "archer"

    def damage(self) -> int:
        return 3


class Knight(Unit):
    def damage(self) -> int:
        return super().damage() + 4


for unit in [Unit(), Archer(), Knight()]:
    print(unit.report())
```

`self.label()` and `self.damage()` are looked up on the actual object each time, so `Archer` swaps in both of its own methods even though `report` lives in the parent. `Knight` keeps the inherited `label` ("unit") and builds its damage on top of the parent's with `super()`.

### Exercise 1: Replace the if/elif chain

Every new shipping option means editing this function, and every caller that checks `method == "express"` somewhere else.

```python title="shipping.py"
def shipping_cost(method: str, weight_kg: float) -> float:
    if method == "standard":
        return 4.99 if weight_kg <= 2 else 4.99 + 1.50 * (weight_kg - 2)
    elif method == "express":
        return 12.00 + 2.00 * weight_kg
    elif method == "pickup":
        return 0.0
    else:
        raise ValueError(f"unknown method {method!r}")


for method in ["standard", "express", "pickup"]:
    print(method, shipping_cost(method, 3.0))
```

Rewrite it as classes `Standard`, `Express`, and `Pickup`, each with a `name` attribute and a `cost(weight_kg)` method. Then add a fourth class, `Drone` (a flat $8.00), and a function `cheapest(methods, weight_kg)` that returns the method with the lowest cost, all without any `if` on the method's name. Expected output for 3 kg:

```text
standard: $6.49
express: $18.00
pickup: $0.00
drone: $8.00
cheapest: pickup
```

:::hint
`min()` takes a `key=` function: `min(methods, key=lambda m: m.cost(weight_kg))` asks each object for its own cost.
:::

:::solution
```python run title="shipping.py"
class Standard:
    name = "standard"

    def cost(self, weight_kg: float) -> float:
        extra = max(0.0, weight_kg - 2)
        return 4.99 + 1.50 * extra


class Express:
    name = "express"

    def cost(self, weight_kg: float) -> float:
        return 12.00 + 2.00 * weight_kg


class Pickup:
    name = "pickup"

    def cost(self, weight_kg: float) -> float:
        return 0.0


class Drone:
    name = "drone"

    def cost(self, weight_kg: float) -> float:
        return 8.00


def cheapest(methods: list, weight_kg: float):
    return min(methods, key=lambda method: method.cost(weight_kg))


methods = [Standard(), Express(), Pickup(), Drone()]
for method in methods:
    print(f"{method.name}: ${method.cost(3.0):.2f}")
print("cheapest:", cheapest(methods, 3.0).name)
```

Adding `Drone` touched nothing else, and `cheapest` works for options that don't exist yet. The classes don't even share a parent; duck typing is enough. An ABC would be worth adding once there are several people writing new options and you want a missing `cost` caught early.
:::

### Exercise 2: One interface, two storage backends

A game saves high scores. Today it uses a dict; next month it should use files. Write an ABC called `Storage` with abstract methods `save(key, value)` and `load(key)`, where `load` raises `KeyError` for a missing key, and a shared, concrete method `load_or(key, default)`. Implement `MemoryStorage` (a dict) and `FileStorage` (one `.txt` file per key in a folder). This code must work unchanged with both:

```python title="scores.py"
def remember_high_score(storage, score: int) -> None:
    best = int(storage.load_or("high_score", "0"))
    if score > best:
        storage.save("high_score", str(score))
```

```mermaid caption="Callers depend on Storage; either backend can be plugged in"
classDiagram
    class Storage {
        <<abstract>>
        +save(key, value)*
        +load(key)* str
        +load_or(key, default) str
    }
    Storage <|-- MemoryStorage
    Storage <|-- FileStorage
    class FileStorage {
        +Path folder
    }
```

:::hint
Reading a missing file raises `FileNotFoundError`, but the interface promises `KeyError`. Catch one and raise the other with `from`, or `load_or` will break for `FileStorage` only.
:::

:::solution
```python run title="storage.py"
from abc import ABC, abstractmethod
from pathlib import Path


class Storage(ABC):
    @abstractmethod
    def save(self, key: str, value: str) -> None: ...

    @abstractmethod
    def load(self, key: str) -> str:
        """Return the value, or raise KeyError if the key is missing."""

    def load_or(self, key: str, default: str) -> str:
        try:
            return self.load(key)
        except KeyError:
            return default


class MemoryStorage(Storage):
    def __init__(self) -> None:
        self._data: dict[str, str] = {}

    def save(self, key: str, value: str) -> None:
        self._data[key] = value

    def load(self, key: str) -> str:
        return self._data[key]


class FileStorage(Storage):
    def __init__(self, folder: Path) -> None:
        self.folder = folder
        folder.mkdir(exist_ok=True)

    def save(self, key: str, value: str) -> None:
        (self.folder / f"{key}.txt").write_text(value)

    def load(self, key: str) -> str:
        try:
            return (self.folder / f"{key}.txt").read_text()
        except FileNotFoundError as err:
            raise KeyError(key) from err


def remember_high_score(storage: Storage, score: int) -> None:
    best = int(storage.load_or("high_score", "0"))
    if score > best:
        storage.save("high_score", str(score))


for storage in [MemoryStorage(), FileStorage(Path("saves"))]:
    for score in [40, 75, 60]:
        remember_high_score(storage, score)
    print(type(storage).__name__, storage.load("high_score"))
```

`load_or` is written once in the ABC and works for both, *because* both keep the same promise about `KeyError`. A backend that raised a different exception would technically have a `load` method yet still break callers. Matching the method names isn't enough; the behavior has to match too.
:::

### Exercise 3: Let the tests find the bug

This function should turn a title into a URL-friendly slug. Write at least four `unittest` tests from the examples in its docstring, run them, and fix whatever they reveal.

```python title="slugify.py"
def slugify(title: str) -> str:
    """Examples:
    'Hello' -> 'hello'
    'Hello, World!' -> 'hello-world'
    '  Big   News  ' -> 'big-news'
    'Top 10 Tips' -> 'top-10-tips'
    """
    cleaned = "".join(ch if ch.isalnum() else " " for ch in title.lower())
    return cleaned.strip().replace(" ", "-")
```

:::hint
Two of the four examples fail. Look at what happens to `", "` in `'Hello, World!'`: two characters that aren't letters, so two spaces.
:::

:::solution
```python run title="test_slugify.py"
import sys
import unittest


def slugify(title: str) -> str:
    cleaned = "".join(ch if ch.isalnum() else " " for ch in title.lower())
    return "-".join(cleaned.split())


class TestSlugify(unittest.TestCase):
    def test_single_word_is_lowercased(self):
        self.assertEqual(slugify("Hello"), "hello")

    def test_punctuation_becomes_one_hyphen(self):
        self.assertEqual(slugify("Hello, World!"), "hello-world")

    def test_runs_of_spaces_collapse(self):
        self.assertEqual(slugify("  Big   News  "), "big-news")

    def test_digits_are_kept(self):
        self.assertEqual(slugify("Top 10 Tips"), "top-10-tips")


suite = unittest.defaultTestLoader.loadTestsFromTestCase(TestSlugify)
unittest.TextTestRunner(stream=sys.stdout, verbosity=2).run(suite)
```

With the original code, the punctuation and spaces tests failed with `'hello--world' != 'hello-world'` and `'big---news' != 'big-news'`. `str.split()` with no argument splits on any run of whitespace and drops empty pieces, so joining with `"-"` gives exactly one hyphen between words.
:::

### Exercise 4: Test with a fake clock and a fake notifier

`ReminderService` texts guests whose check-in is tomorrow. Testing it with the real date would give different results every day. Write a `FakeClock` with a `today()` method that returns a fixed date, a `FakeNotifier` that records messages, and three tests using `setUp`: only tomorrow's guests get a message, `send_reminders` returns how many were sent, and an empty list sends nothing.

```python title="reminders.py"
from datetime import date, timedelta


class ReminderService:
    def __init__(self, clock, notifier) -> None:
        self.clock = clock
        self.notifier = notifier

    def send_reminders(self, bookings: list[tuple[str, date]]) -> int:
        tomorrow = self.clock.today() + timedelta(days=1)
        sent = 0
        for guest, check_in in bookings:
            if check_in == tomorrow:
                self.notifier.send(guest, f"See you tomorrow, {check_in}!")
                sent += 1
        return sent
```

:::hint
The service never asks *what kind* of clock it has, only that it has `today()`. Your fake can be three lines long. Make `setUp` build the clock, the notifier, and the service.
:::

:::solution
```python run title="test_reminders.py"
import sys
import unittest
from datetime import date, timedelta


class ReminderService:
    def __init__(self, clock, notifier) -> None:
        self.clock = clock
        self.notifier = notifier

    def send_reminders(self, bookings: list[tuple[str, date]]) -> int:
        tomorrow = self.clock.today() + timedelta(days=1)
        sent = 0
        for guest, check_in in bookings:
            if check_in == tomorrow:
                self.notifier.send(guest, f"See you tomorrow, {check_in}!")
                sent += 1
        return sent


class FakeClock:
    def __init__(self, fixed: date) -> None:
        self.fixed = fixed

    def today(self) -> date:
        return self.fixed


class FakeNotifier:
    def __init__(self) -> None:
        self.sent: list[tuple[str, str]] = []

    def send(self, to: str, text: str) -> None:
        self.sent.append((to, text))


class TestReminders(unittest.TestCase):
    def setUp(self):
        self.notifier = FakeNotifier()
        self.service = ReminderService(FakeClock(date(2026, 12, 1)), self.notifier)
        self.bookings = [("Ana", date(2026, 12, 2)), ("Ben", date(2026, 12, 5)), ("Cy", date(2026, 12, 2))]

    def test_only_tomorrows_guests_are_messaged(self):
        self.service.send_reminders(self.bookings)
        self.assertEqual([to for to, _ in self.notifier.sent], ["Ana", "Cy"])

    def test_returns_number_sent(self):
        self.assertEqual(self.service.send_reminders(self.bookings), 2)

    def test_no_bookings_sends_nothing(self):
        self.assertEqual(self.service.send_reminders([]), 0)
        self.assertEqual(self.notifier.sent, [])


suite = unittest.defaultTestLoader.loadTestsFromTestCase(TestReminders)
unittest.TextTestRunner(stream=sys.stdout, verbosity=2).run(suite)
```

Because the clock is passed in rather than created inside the class with `date.today()`, the tests control time. That's a design choice you make for testability, and it costs one constructor parameter.

### Exercise 5: Make the spec pass

These tests are the whole specification for a club `Waitlist`. Write `Waitlist`, `AlreadyWaiting`, and `EmptyWaitlist` so that all five pass. Don't change the tests.

```python title="test_waitlist.py"
import unittest


class TestWaitlist(unittest.TestCase):
    def setUp(self):
        self.waitlist = Waitlist()
        for name in ["Ana", "Ben", "Cy"]:
            self.waitlist.join(name)

    def test_positions_start_at_one(self):
        self.assertEqual(self.waitlist.position("Cy"), 3)

    def test_next_serves_the_first_to_join(self):
        self.assertEqual(self.waitlist.next(), "Ana")
        self.assertEqual(self.waitlist.position("Ben"), 1)

    def test_len_counts_people_waiting(self):
        self.assertEqual(len(self.waitlist), 3)

    def test_joining_twice_is_refused(self):
        with self.assertRaises(AlreadyWaiting):
            self.waitlist.join("Ben")
        self.assertEqual(len(self.waitlist), 3)

    def test_next_on_empty_waitlist_raises(self):
        with self.assertRaises(EmptyWaitlist):
            Waitlist().next()
```

:::hint
Read each test as a sentence about behavior. `len(self.waitlist)` needs `__len__`. The last line of `test_joining_twice_is_refused` checks that a refused join leaves the list unchanged, so raise *before* appending.
:::

:::solution
```python run title="waitlist.py"
import sys
import unittest


class AlreadyWaiting(Exception):
    """That person is already on the waitlist."""


class EmptyWaitlist(Exception):
    """There is nobody to serve."""


class Waitlist:
    def __init__(self) -> None:
        self._names: list[str] = []

    def join(self, name: str) -> None:
        if name in self._names:
            raise AlreadyWaiting(name)
        self._names.append(name)

    def next(self) -> str:
        if not self._names:
            raise EmptyWaitlist("nobody is waiting")
        return self._names.pop(0)

    def position(self, name: str) -> int:
        return self._names.index(name) + 1

    def __len__(self) -> int:
        return len(self._names)


class TestWaitlist(unittest.TestCase):
    def setUp(self):
        self.waitlist = Waitlist()
        for name in ["Ana", "Ben", "Cy"]:
            self.waitlist.join(name)

    def test_positions_start_at_one(self):
        self.assertEqual(self.waitlist.position("Cy"), 3)

    def test_next_serves_the_first_to_join(self):
        self.assertEqual(self.waitlist.next(), "Ana")
        self.assertEqual(self.waitlist.position("Ben"), 1)

    def test_len_counts_people_waiting(self):
        self.assertEqual(len(self.waitlist), 3)

    def test_joining_twice_is_refused(self):
        with self.assertRaises(AlreadyWaiting):
            self.waitlist.join("Ben")
        self.assertEqual(len(self.waitlist), 3)

    def test_next_on_empty_waitlist_raises(self):
        with self.assertRaises(EmptyWaitlist):
            Waitlist().next()


suite = unittest.defaultTestLoader.loadTestsFromTestCase(TestWaitlist)
unittest.TextTestRunner(stream=sys.stdout, verbosity=2).run(suite)
```

The tests decided the method names, the 1-based positions, and the exception classes before any code existed. That's what test-first means in practice: the caller's view of the interface comes first.
:::

### Exercise 6: The same spec in pytest

### Challenge: A password policy you can extend without editing

Write an ABC `Rule` with an abstract method `problem(password) -> str | None` (a message if the rule is broken, `None` if it's fine). Implement `MinLength(n)`, `HasDigit()`, and `NotCommon(words)`. Then write `PasswordPolicy(rules)` with a method `problems(password)` that returns every message. Add a fourth rule, `NoSpaces()`, without editing `PasswordPolicy`, and write `unittest` tests for a strong password, a weak one that breaks three rules, and the new rule.

```mermaid caption="The policy holds rules; it never needs to know which ones"
classDiagram
    class Rule {
        <<abstract>>
        +problem(password)* str or None
    }
    Rule <|-- MinLength
    Rule <|-- HasDigit
    Rule <|-- NotCommon
    Rule <|-- NoSpaces
    PasswordPolicy o-- Rule : rules
```

:::hint
`problems` is a one-line list comprehension: ask every rule for its `problem(password)` and keep the ones that aren't `None`.
:::

:::solution
```python run title="password_policy.py"
import sys
import unittest
from abc import ABC, abstractmethod


class Rule(ABC):
    @abstractmethod
    def problem(self, password: str) -> str | None:
        """Return a message if the password breaks this rule, else None."""


class MinLength(Rule):
    def __init__(self, length: int) -> None:
        self.length = length

    def problem(self, password: str) -> str | None:
        if len(password) < self.length:
            return f"use at least {self.length} characters"
        return None


class HasDigit(Rule):
    def problem(self, password: str) -> str | None:
        return None if any(ch.isdigit() for ch in password) else "include a digit"


class NotCommon(Rule):
    def __init__(self, words: set[str]) -> None:
        self.words = words

    def problem(self, password: str) -> str | None:
        return "that password is too common" if password.lower() in self.words else None


class NoSpaces(Rule):
    def problem(self, password: str) -> str | None:
        return "remove the spaces" if " " in password else None


class PasswordPolicy:
    def __init__(self, rules: list[Rule]) -> None:
        self.rules = rules

    def problems(self, password: str) -> list[str]:
        return [msg for rule in self.rules if (msg := rule.problem(password)) is not None]


class TestPolicy(unittest.TestCase):
    def setUp(self):
        self.policy = PasswordPolicy([MinLength(10), HasDigit(), NotCommon({"password", "letmein"})])

    def test_strong_password_has_no_problems(self):
        self.assertEqual(self.policy.problems("purple-otter-42"), [])

    def test_weak_password_lists_every_problem(self):
        self.assertEqual(len(self.policy.problems("password")), 3)

    def test_new_rule_plugs_in(self):
        policy = PasswordPolicy(self.policy.rules + [NoSpaces()])
        self.assertEqual(policy.problems("purple otter 42"), ["remove the spaces"])


suite = unittest.defaultTestLoader.loadTestsFromTestCase(TestPolicy)
unittest.TextTestRunner(stream=sys.stdout, verbosity=2).run(suite)
```

`PasswordPolicy` never changed when `NoSpaces` arrived; it only depends on the `Rule` interface. That property, open to new behavior without editing old code, has a name you'll meet in the next module: the open/closed principle. The `:=` in `problems` (the "walrus" operator) stores each rule's message so it's computed once and then tested.
:::
