The fastest way to finish a capstone is to have the whole thing working, badly, on the first day, and then improve one piece at a time. This page is that first day. Each milestone below is a file (or a few) in the starter repo, with a cell that proves it works. If you're doing project 1, extend these files. For any other project, keep the structure and swap the domain: your data in `make_data.py`, your features in `FEATURES`, your contract in `app/main.py`.

## M1: Train a candidate and register it

**Done when:** `registry/<name>/aliases.json` names a champion, and its record holds the artifact hash, the data hash, the features, holdout metrics, a golden input with its expected score, and a reference profile for drift.

```python run title="make_data.py, train.py"
import hashlib
import json
import os
import shutil
import sys
import time
import unittest
import uuid
from datetime import datetime, timezone
from pathlib import Path

import joblib
import numpy as np
import pandas as pd
from sklearn.compose import ColumnTransformer
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import recall_score, roc_auc_score
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import OneHotEncoder, StandardScaler

def make_orders(n=2000, seed=23):                     # make_data.py: swap in your own source
    rng = np.random.default_rng(seed)
    distance_km = rng.gamma(shape=2.0, scale=2.0, size=n).round(2)
    items = rng.integers(1, 9, size=n)
    hour = rng.integers(10, 23, size=n)
    raining = (rng.random(n) < 0.25).astype(int)
    rush = (hour >= 17) & (hour <= 19)
    logit = -4.0 + 0.45 * distance_km + 0.15 * items + 1.2 * raining + 0.9 * rush
    late = (rng.random(n) < 1 / (1 + np.exp(-logit))).astype(int)
    return pd.DataFrame({"distance_km": distance_km, "items": items,
                         "hour": hour, "raining": raining, "late": late})

NAME, FEATURES, THRESHOLD = "late-delivery", ["distance_km", "items", "hour", "raining"], 0.5
GOLDEN = {"distance_km": 6.5, "items": 3, "hour": 18, "raining": 1}
Path("data").mkdir(exist_ok=True)
make_orders().to_csv("data/orders.csv", index=False)
make_orders(1000, seed=99).to_csv("data/holdout.csv", index=False)

def sha256(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def profile(frame):
    out = {}
    for col in ["distance_km", "items", "hour"]:
        edges = np.unique(np.quantile(frame[col], np.linspace(0, 1, 11)))
        edges[0], edges[-1] = -np.inf, np.inf
        counts = np.histogram(frame[col], edges)[0]
        out[col] = {"edges": edges.tolist(), "fractions": (counts / counts.sum()).tolist()}
    return out

def train(csv_path="data/orders.csv", out="candidate"):
    data = pd.read_csv(csv_path)
    prep = ColumnTransformer([("scale", StandardScaler(), ["distance_km", "items"]),
                              ("hour", OneHotEncoder(handle_unknown="ignore"), ["hour"])],
                             remainder="passthrough")
    model = Pipeline([("prep", prep), ("clf", LogisticRegression(max_iter=1000))]).fit(data[FEATURES], data["late"])
    holdout = pd.read_csv("data/holdout.csv")
    p = model.predict_proba(holdout[FEATURES])[:, 1]
    shutil.rmtree(out, ignore_errors=True)
    Path(out).mkdir()
    joblib.dump(model, f"{out}/model.joblib")
    record = {"name": NAME, "git_sha": os.environ.get("GITHUB_SHA", "local")[:7],
              "training_data": {"file": csv_path, "sha256": sha256(csv_path)},
              "schema": {"features": FEATURES, "threshold": THRESHOLD},
              "metrics": {"holdout_roc_auc": round(roc_auc_score(holdout.late, p), 4),
                          "holdout_recall": round(recall_score(holdout.late, p >= THRESHOLD), 4)},
              "golden": {"input": GOLDEN, "probability": float(model.predict_proba(pd.DataFrame([GOLDEN]))[0, 1])},
              "reference": profile(data)}
    Path(f"{out}/record.json").write_text(json.dumps(record, indent=2))
    return record

print(json.dumps(train()["metrics"]))
```

```python run title="registry.py"
# In the repo, registry.py imports json, shutil, hashlib, joblib, and Path itself,
# and defines NAME and sha256(); on this page they come from the cell above.
REGISTRY = Path("registry") / NAME

def read_aliases():
    path = REGISTRY / "aliases.json"
    return json.loads(path.read_text()) if path.exists() else {"current": {}, "history": []}

def set_alias(alias, version, reason):
    if not (REGISTRY / version / "record.json").exists():
        raise ValueError(f"{version} is not registered")
    aliases = read_aliases()
    aliases["history"].append({"alias": alias, "from": aliases["current"].get(alias), "to": version,
                               "reason": reason, "at": datetime.now(timezone.utc).isoformat(timespec="seconds")})
    aliases["current"][alias] = version
    (REGISTRY / "aliases.json").write_text(json.dumps(aliases, indent=2))

def register(candidate, version):
    final = REGISTRY / version
    if final.exists():
        raise FileExistsError(f"{version} is already registered; versions are immutable")
    shutil.copytree(candidate, final)
    record = json.loads((final / "record.json").read_text())
    record.update(version=version, artifact_sha256=sha256(final / "model.joblib"),
                  parent=read_aliases()["current"].get("champion"))
    (final / "record.json").write_text(json.dumps(record, indent=2))
    return record

def load_alias(alias):
    version = read_aliases()["current"][alias]
    record = json.loads((REGISTRY / version / "record.json").read_text())
    if sha256(REGISTRY / version / "model.joblib") != record["artifact_sha256"]:
        raise RuntimeError(f"{version}: artifact does not match its record; refusing to load")
    return joblib.load(REGISTRY / version / "model.joblib"), record

def rollback(alias, reason):
    previous = [m for m in read_aliases()["history"] if m["alias"] == alias][-1]["from"]
    if previous is None:
        raise ValueError("nothing to roll back to")
    set_alias(alias, previous, f"rollback: {reason}")
    return previous

shutil.rmtree("registry", ignore_errors=True)
register("candidate", "1.0.0")
set_alias("champion", "1.0.0", "first production model")
print(read_aliases()["current"], "|", sorted(p.name for p in (REGISTRY / "1.0.0").iterdir()))
```

These are compact versions of your Module 4 functions. Promotion is deliberately a separate, manual step: CI trains and gates a candidate, but a person runs `register` and `set_alias` and commits the registry change in a pull request, so every champion change is reviewed and appears in Git history.

## M2: Serve the champion with honest health checks

**Done when:** `/readyz` returns 503 before the model loads and 200 after; it reports the version and verifies the golden input; every `/predict` writes one JSON log line, including rejected requests.

```python run title="app/main.py"
# In the repo this file also imports its helpers:
#   from registry import load_alias, NAME, FEATURES, THRESHOLD
import logging
from contextlib import asynccontextmanager

from fastapi import FastAPI, HTTPException, Request, Response
from fastapi.testclient import TestClient
from pydantic import BaseModel, ConfigDict, Field

log = logging.getLogger("predictions")
log.setLevel(logging.INFO)
log.handlers = [logging.FileHandler(os.environ.get("PREDICTION_LOG", "predictions.jsonl"), mode="w")]
log.propagate = False

class Order(BaseModel):
    model_config = ConfigDict(extra="forbid")
    distance_km: float = Field(gt=0, le=50)
    items: int = Field(ge=1, le=50)
    hour: int = Field(ge=0, le=23)
    raining: bool

class Prediction(BaseModel):
    late_probability: float
    late: bool
    model_version: str

state = {}

@asynccontextmanager
async def lifespan(app):
    state["model"], state["record"] = load_alias("champion")
    yield
    state.clear()

app = FastAPI(title=NAME, lifespan=lifespan)

@app.middleware("http")
async def log_predictions(request: Request, call_next):
    start = time.perf_counter()
    response = await call_next(request)
    if request.url.path == "/predict":
        line = {"request_id": request.headers.get("x-request-id") or uuid.uuid4().hex[:12],
                "status": response.status_code, "latency_ms": round((time.perf_counter() - start) * 1000, 2),
                "model_version": state.get("record", {}).get("version"), **getattr(request.state, "logged", {})}
        log.info(json.dumps(line))
    return response

@app.get("/livez")
def livez():
    return {"status": "alive"}

@app.get("/readyz")
def readyz(response: Response):
    problems = []
    if "model" not in state:
        problems.append("model not loaded")
    else:
        record, expected = state["record"], os.environ.get("EXPECTED_MODEL_VERSION")
        if expected and record["version"] != expected:
            problems.append(f"serving {record['version']}, expected {expected}")
        score = state["model"].predict_proba(pd.DataFrame([record["golden"]["input"]]))[0, 1]
        if abs(score - record["golden"]["probability"]) > 1e-6:
            problems.append("golden input scored differently than at training time")
    if problems:
        response.status_code = 503
        return {"status": "not ready", "problems": problems}
    return {"status": "ready", "model_version": state["record"]["version"],
            "artifact_sha256": state["record"]["artifact_sha256"][:12]}

@app.post("/predict", response_model=Prediction)
def predict(order: Order, request: Request):
    if "model" not in state:
        raise HTTPException(status_code=503, detail="model not loaded")
    row = pd.DataFrame([order.model_dump()], columns=FEATURES)
    p = float(state["model"].predict_proba(row)[0, 1])
    request.state.logged = {**order.model_dump(), "late_probability": round(p, 4)}
    return Prediction(late_probability=round(p, 3), late=p >= THRESHOLD, model_version=state["record"]["version"])

print("before startup:", TestClient(app).get("/readyz").status_code)
with TestClient(app) as client:
    print("after startup: ", client.get("/readyz").json())
    print("predict:       ", client.post("/predict", json={"distance_km": 6.5, "items": 3, "hour": 18, "raining": True}).json())
    print("bad input:     ", client.post("/predict", json={"distance_km": 6.5, "items": 3, "hour": 25, "raining": True}).status_code)
print("log lines:", len(Path("predictions.jsonl").read_text().splitlines()))
```

## M3: Tests and the gate

**Done when:** `python -m pytest` passes on a good candidate, and `python gate.py` exits 1 for a candidate trained on broken labels.

```python run title="tests/ and gate.py"
class PageTest(unittest.FunctionTestCase):
    def __str__(self):
        return self.id()

def run_tests(*tests):
    suite = unittest.TestSuite(PageTest(t) for t in tests)
    return unittest.TextTestRunner(stream=sys.stdout, verbosity=2).run(suite).wasSuccessful()

def test_data_schema_and_ranges():                    # tests/test_data.py
    data = pd.read_csv("data/orders.csv")
    assert list(data.columns) == FEATURES + ["late"]
    bad = data.query("hour < 0 or hour > 23 or distance_km <= 0 or distance_km > 50")
    assert bad.empty, f"{len(bad)} rows out of range"

def test_rain_raises_risk():                          # tests/test_model.py
    model = joblib.load("candidate/model.joblib")
    dry, wet = pd.DataFrame([{**GOLDEN, "raining": 0}]), pd.DataFrame([{**GOLDEN, "raining": 1}])
    assert model.predict_proba(wet)[0, 1] > model.predict_proba(dry)[0, 1]

def test_contract():                                  # tests/test_api.py
    with TestClient(app) as client:
        body = client.post("/predict", json={"distance_km": 2.0, "items": 1, "hour": 12, "raining": False}).json()
        assert set(body) == {"late_probability", "late", "model_version"}
        assert client.post("/predict", json={"distance_km": 2.0}).status_code == 422

def gate(max_drop=0.01, floors={"holdout_roc_auc": 0.75, "holdout_recall": 0.40}):   # gate.py
    new = json.loads(Path("candidate/record.json").read_text())["metrics"]
    _, champ = load_alias("champion")
    failures = [f"{m} {new[m]} below {f}" for m, f in floors.items() if new[m] < f]
    if new["holdout_roc_auc"] < champ["metrics"]["holdout_roc_auc"] - max_drop:
        failures.append(f"AUC {new['holdout_roc_auc']} vs champion {champ['metrics']['holdout_roc_auc']}")
    print("GATE", "FAILED: " + "; ".join(failures) if failures else "PASSED")
    return 1 if failures else 0

run_tests(test_data_schema_and_ranges, test_rain_raises_risk, test_contract)
print("exit code:", gate())

broken = pd.read_csv("data/orders.csv")
broken["late"] = broken["late"].sample(frac=1, random_state=3).values
broken.to_csv("data/broken.csv", index=False)
train("data/broken.csv")
print("exit code:", gate())
train()                                              # restore the good candidate
```

Your project needs more tests than these three; Module 5's lessons have the full set (data, behavior, contract snapshot, slices). These are the minimum that prove the wiring.

## M4: Ship it

**Done when:** `docker build` succeeds, the container reports `healthy` in `docker ps`, and a push to `main` runs CI to a green check (and, if you deploy, the smoke test confirms the new version).

```dockerfile title="Dockerfile"
FROM python:3.12-slim

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PREDICTION_LOG=/tmp/predictions.jsonl

WORKDIR /app
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY app/ app/
COPY registry.py .
COPY registry/ registry/

RUN useradd --create-home appuser
USER appuser
EXPOSE 8000

HEALTHCHECK --interval=30s --timeout=3s --start-period=15s --retries=3 \
  CMD python -c "import urllib.request; urllib.request.urlopen('http://127.0.0.1:8000/readyz', timeout=2)"

CMD ["sh", "-c", "exec uvicorn app.main:app --host 0.0.0.0 --port ${PORT:-8000}"]
```

The health check calls `/readyz`, so the container only turns healthy once the right model is loaded and scores its golden input correctly. In production, point `PREDICTION_LOG` at nothing and log to stdout instead, so the platform collects the lines; the file here keeps the demo self-contained.

```yaml title=".github/workflows/ci.yml"
name: ci

on:
  push:
    branches: [main]
  pull_request:

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7
      - uses: actions/setup-python@v7
        with:
          python-version: "3.12"
          cache: pip
          cache-dependency-path: requirements*.txt
      - run: pip install -r requirements-dev.txt ruff
      - run: ruff check .
      - run: python make_data.py
      - run: python -m pytest tests/test_data.py -v
      - run: python train.py
      - run: python -m pytest -v
      - run: python gate.py
      - run: docker build -t capstone:${{ github.sha }} .

  deploy:
    needs: test
    if: github.event_name == 'push' && github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    environment: production
    steps:
      - uses: actions/checkout@v7
      - run: curl -fsS -X POST "${{ secrets.RENDER_DEPLOY_HOOK_URL }}&ref=${{ github.sha }}"
      - name: Smoke test the champion version
        run: |
          want=$(python -c "import json; print(json.load(open('registry/late-delivery/aliases.json'))['current']['champion'])")
          for attempt in $(seq 1 30); do
            got=$(curl -fsS "${{ vars.SERVICE_URL }}/readyz" | python -c "import sys, json; print(json.load(sys.stdin)['model_version'])") || true
            [ "$got" = "$want" ] && echo "serving $got" && exit 0
            sleep 20
          done
          exit 1
```

The `docker build` step in `test` proves the image builds on every pull request, before anything deploys. If you don't deploy, delete the `deploy` job; everything else still applies.

## M5: Watch it

**Done when:** `monitor.py` reads the prediction log and outcomes and prints one status per signal, with the overall status at the bottom.

```python run title="monitor.py"
def psi(ref, values):
    edges, expected = np.array(ref["edges"]), np.clip(np.array(ref["fractions"]), 1e-4, None)
    counts = np.histogram(values, edges)[0]
    actual = np.clip(counts / counts.sum(), 1e-4, None)
    return float(np.sum((actual - expected) * np.log(actual / expected)))

RULES = [("p99 latency ms", 250, "PAGE"), ("5xx rate", 0.01, "PAGE"), ("4xx rate", 0.05, "TICKET"),
         ("PSI distance_km", 0.25, "TICKET"), ("late-rate gap", 0.10, "PAGE")]

def monitor(log_path, outcomes, record):
    log = pd.read_json(log_path, lines=True)
    ok = log[log.status == 200].merge(outcomes, on="request_id", how="left")
    values = {"p99 latency ms": float(np.percentile(ok.latency_ms, 99)),
              "5xx rate": float((log.status >= 500).mean()),
              "4xx rate": float(log.status.between(400, 499).mean()),
              "PSI distance_km": psi(record["reference"]["distance_km"], ok.distance_km),
              "late-rate gap": float(ok.late.mean() - ok.late_probability.mean())}
    worst = "OK"
    for signal, limit, severity in RULES:
        status = severity if values[signal] > limit else "OK"
        worst = "PAGE" if "PAGE" in (worst, status) else ("TICKET" if "TICKET" in (worst, status) else "OK")
        print(f"{signal:16}{values[signal]:9.3f}{limit:9.3f}  {status}")
    print("overall:", worst)
    return worst

day = make_orders(600, seed=101)
day["request_id"] = [f"ord-{i}" for i in range(len(day))]
with TestClient(app) as client:
    for row in day.itertuples():
        client.post("/predict", headers={"X-Request-ID": row.request_id},
                    json={"distance_km": row.distance_km, "items": int(row.items), "hour": int(row.hour),
                          "raining": bool(row.raining)})
monitor("predictions.jsonl", day[["request_id", "late"]], load_alias("champion")[1])
```

On a normal day every signal is OK. To check the monitor can fail, replay Module 6's afternoon (a new zone and a road closure) and make sure it pages; a monitor that has never fired hasn't been tested.

## M6: Release, fail, and recover on purpose

**Done when:** you have run this sequence against your own service and recorded how long the rollback took: register a new version, promote it, see it misbehave, roll back, and confirm the old version is serving.

```python run title="release.py (a rollback drill)"
outage = make_orders(2000, seed=30)
outage.loc[outage.raining == 1, "raining"] = 0          # weather feed down during training
outage.to_csv("data/orders_outage.csv", index=False)
train("data/orders_outage.csv")
register("candidate", "1.1.0")
set_alias("champion", "1.1.0", "drill: promote a known-bad candidate")

with TestClient(app) as client:
    rainy = {"distance_km": 4.0, "items": 3, "hour": 18, "raining": True}
    print("serving", client.get("/readyz").json()["model_version"],
          "| rainy order:", client.post("/predict", json=rainy).json()["late_probability"])

start = time.perf_counter()
print("rolled back to", rollback("champion", "drill: rainy orders under-warned"))
with TestClient(app) as client:                         # a restart: lifespan reloads the alias
    print("serving", client.get("/readyz").json()["model_version"],
          "| rainy order:", client.post("/predict", json=rainy).json()["late_probability"])
print(f"rollback and reload took {time.perf_counter() - start:.2f} s on this machine")
for move in read_aliases()["history"]:
    print(f"  {move['from']} -> {move['to']}: {move['reason']}")
```

The drill exercises the registry, the alias, the reload, and your runbook's last step ("confirm the old version is serving") in one go. On a real host the slow part is the restart, so time the drill there too, and write the number in `RUNBOOK.md`. If any step needed a command you had to look up, add it to the runbook now, while you remember.

## Where to go from here

With all six milestones working, spend the rest of your time on depth in the requirement that's weakest for your project: more slices in the gate, a real canary router, a better model through the same gates, or a deployed service with the CD job.
