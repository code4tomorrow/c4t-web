## Warm-up: what does a loose schema let through?

Run this setup cell first. It rebuilds the course model and starts with a deliberately loose service: the request model has types but no ranges.

```python run
import json
import time
from contextlib import asynccontextmanager
from pathlib import Path

import joblib
import numpy as np
import pandas as pd
import sklearn
from fastapi import FastAPI, HTTPException
from fastapi.testclient import TestClient
from pydantic import BaseModel, ConfigDict, Field
from sklearn.compose import ColumnTransformer
from sklearn.linear_model import LogisticRegression
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import OneHotEncoder, StandardScaler

rng = np.random.default_rng(23)
n = 2000
orders = pd.DataFrame({
    "distance_km": rng.gamma(2.0, 2.0, n).round(2),
    "items": rng.integers(1, 9, n),
    "hour": rng.integers(10, 23, n),
    "raining": (rng.random(n) < 0.25).astype(int),
})
rush = orders["hour"].between(17, 19)
logit = -4.0 + 0.45 * orders["distance_km"] + 0.15 * orders["items"] + 1.2 * orders["raining"] + 0.9 * rush
orders["late"] = (rng.random(n) < 1 / (1 + np.exp(-logit))).astype(int)

FEATURES = ["distance_km", "items", "hour", "raining"]
model = Pipeline([
    ("prep", ColumnTransformer([
        ("scale", StandardScaler(), ["distance_km", "items"]),
        ("hour", OneHotEncoder(handle_unknown="ignore"), ["hour"]),
    ], remainder="passthrough")),
    ("clf", LogisticRegression(max_iter=1000)),
]).fit(orders[FEATURES], orders["late"])

Path("models").mkdir(exist_ok=True)
joblib.dump(model, "models/model.joblib")
Path("models/model.json").write_text(json.dumps(
    {"version": "1.0.0", "features": FEATURES, "threshold": 0.5, "sklearn_version": sklearn.__version__}))

state = {}

@asynccontextmanager
async def lifespan(app):
    state["model"] = joblib.load("models/model.joblib")
    state["meta"] = json.loads(Path("models/model.json").read_text())
    yield
    state.clear()

def score(rows):
    """Late probabilities for a list of order dicts, in one predict_proba call."""
    frame = pd.DataFrame(rows, columns=state["meta"]["features"])
    return state["model"].predict_proba(frame)[:, 1]

class LooseOrder(BaseModel):
    distance_km: float
    items: int
    hour: int
    raining: bool

loose_app = FastAPI(lifespan=lifespan)

@loose_app.post("/predict")
def loose_predict(order: LooseOrder):
    return {"late_probability": round(float(score([order.model_dump()])[0]), 3)}

print("setup done")
```

Before running the next cell, write down the status code you expect for each request.

```python run
requests_to_try = {
    "normal order":      {"distance_km": 6.5, "items": 3, "hour": 18, "raining": True},
    "negative distance": {"distance_km": -3, "items": 3, "hour": 18, "raining": True},
    "hour 40":           {"distance_km": 6.5, "items": 3, "hour": 40, "raining": True},
    "missing raining":   {"distance_km": 6.5, "items": 3, "hour": 18},
    "items as a word":   {"distance_km": 6.5, "items": "three", "hour": 18, "raining": True},
}
with TestClient(loose_app) as client:
    for label, body in requests_to_try.items():
        response = client.post("/predict", json=body)
        print(f"{label:18} {response.status_code}  {response.json() if response.status_code == 200 else ''}")
```

Types alone caught two problems. The other two are worse than errors: a delivery to a negative distance and an order placed at hour 40 both got confident-looking probabilities. The `OneHotEncoder` ignored the unknown hour 40, and the scaler happily scaled −3 km. A model always returns a number, so it's the contract's job to refuse inputs that make no sense.

### Exercise 1: Tighten the contract

Write an `Order` model that rejects all three bad cases from the warm-up (negative distance, hour 40, and an unknown field such as `"tip"`), and a `strict_app` that uses it. Allow distances above 0 and up to 50 km, 1 to 50 items, and hours 0 to 23.

```text
normal order       200
negative distance  422  ['body', 'distance_km']
hour 40            422  ['body', 'hour']
extra field        422  ['body', 'tip']
```

:::hint
`Field(gt=0, le=50)` means "greater than 0, at most 50". Unknown fields are allowed by default; one line in `model_config` changes that. Each 422 body has a `detail` list whose items carry a `loc`.
:::

:::solution
```python run
class Order(BaseModel):
    model_config = ConfigDict(extra="forbid")
    distance_km: float = Field(gt=0, le=50)
    items: int = Field(ge=1, le=50)
    hour: int = Field(ge=0, le=23)
    raining: bool

strict_app = FastAPI(lifespan=lifespan)

@strict_app.post("/predict")
def strict_predict(order: Order):
    return {"late_probability": round(float(score([order.model_dump()])[0]), 3)}

cases = {
    "normal order":      requests_to_try["normal order"],
    "negative distance": requests_to_try["negative distance"],
    "hour 40":           requests_to_try["hour 40"],
    "extra field":       {**requests_to_try["normal order"], "tip": 2},
}
with TestClient(strict_app) as client:
    for label, body in cases.items():
        response = client.post("/predict", json=body)
        where = response.json()["detail"][0]["loc"] if response.status_code == 422 else ""
        print(f"{label:18} {response.status_code}  {where}")
```

The constraints live in the schema, so FastAPI rejects bad requests before `strict_predict` runs, and the `/docs` page shows the allowed ranges to every client developer.
:::

### Exercise 2: Make the health check tell the truth

Add a `GET /health` to a new app, `health_app`, that returns `{"status": "ok", "model_version": ...}` when the model is loaded and **503** when it isn't. Prove both branches: call it once inside `with TestClient(...)` and once without `with`.

```text
with lifespan:    200 {'status': 'ok', 'model_version': '1.0.0'}
without lifespan: 503 {'detail': 'model not loaded'}
```

:::hint
`raise HTTPException(status_code=503, detail="...")` produces the error response. The lifespan only runs when the test client is used as a context manager.
:::

:::solution
```python run
health_app = FastAPI(lifespan=lifespan)

@health_app.get("/health")
def health():
    if "model" not in state:
        raise HTTPException(status_code=503, detail="model not loaded")
    return {"status": "ok", "model_version": state["meta"]["version"]}

with TestClient(health_app) as client:
    response = client.get("/health")
    print("with lifespan:   ", response.status_code, response.json())

response = TestClient(health_app).get("/health")
print("without lifespan:", response.status_code, response.json())
```

Load balancers and Docker's `HEALTHCHECK` stop sending traffic to an instance whose health check fails. A check that only proves "the process is running" would keep routing requests to a server that can't predict.
:::

### Exercise 3: Score many orders in one request

A partner wants to send up to 100 orders at once. Add `POST /predict/batch` to `strict_app` taking `{"orders": [...]}` (1 to 100 `Order` objects) and returning one probability per order, in order, computed with a **single** `predict_proba` call. Requests with 0 or 101 orders must get 422.

```text
3 orders   -> 200 [0.825, 0.029, 0.427]
0 orders   -> 422
101 orders -> 422
```

:::hint
A Pydantic model can have a field of type `list[Order]`, and `Field(min_length=1, max_length=100)` limits the list's length. `score()` already accepts a list of dicts.
:::

:::solution
```python run
class Batch(BaseModel):
    model_config = ConfigDict(extra="forbid")
    orders: list[Order] = Field(min_length=1, max_length=100)

@strict_app.post("/predict/batch")
def predict_batch(batch: Batch):
    probabilities = score([order.model_dump() for order in batch.orders])
    return {"late_probabilities": [round(float(p), 3) for p in probabilities]}

three = [requests_to_try["normal order"],
         {"distance_km": 1.2, "items": 1, "hour": 11, "raining": False},
         {"distance_km": 4.0, "items": 6, "hour": 18, "raining": False}]
with TestClient(strict_app) as client:
    response = client.post("/predict/batch", json={"orders": three})
    print("3 orders   ->", response.status_code, response.json()["late_probabilities"])
    print("0 orders   ->", client.post("/predict/batch", json={"orders": []}).status_code)
    print("101 orders ->", client.post("/predict/batch", json={"orders": three[:1] * 101}).status_code)
```

One `predict_proba` call on 100 rows costs barely more than one call on 1 row, because the fixed overhead (building a DataFrame, running each pipeline step) is paid once. The size limit protects the server: without it, one request with a million orders could hold a worker for minutes. Module 3 measures this trade-off properly.
:::

### Exercise 4: Log one JSON line per request

When something goes wrong in production, you search logs. Build `logged_app`, a copy of `strict_app`'s `/predict` with HTTP middleware that, for every request, writes one JSON line with a request ID, method, path, status code, and duration in milliseconds, and returns the ID in an `X-Request-ID` response header. If the client sent its own `X-Request-ID`, reuse it so one ID can follow a request across services.

```text
{"request_id": "order-1234", "method": "POST", "path": "/predict", "status": 200, "duration_ms": 3.1}
{"request_id": "5f1c0e9a2b7d", "method": "POST", "path": "/predict", "status": 422, "duration_ms": 1.4}
header: order-1234
```

:::hint
`@logged_app.middleware("http")` decorates an `async def` that receives `request` and `call_next`; `response = await call_next(request)` runs the endpoint. Add middleware to a fresh app: Starlette raises `RuntimeError: Cannot add middleware after an application has started` if the app has already served a request, which `strict_app` has. Use `logging` with a `StreamHandler(sys.stdout)` rather than `print`, so the log level and destination can change without code edits.
:::

:::solution
```python run
import logging
import sys
import uuid

log = logging.getLogger("late_delivery")
log.setLevel(logging.INFO)
log.handlers = [logging.StreamHandler(sys.stdout)]

logged_app = FastAPI(lifespan=lifespan)

@logged_app.post("/predict")
def logged_predict(order: Order):
    return {"late_probability": round(float(score([order.model_dump()])[0]), 3)}

@logged_app.middleware("http")
async def log_requests(request, call_next):
    request_id = request.headers.get("x-request-id") or uuid.uuid4().hex[:12]
    start = time.perf_counter()
    response = await call_next(request)
    response.headers["X-Request-ID"] = request_id
    log.info(json.dumps({
        "request_id": request_id, "method": request.method, "path": request.url.path,
        "status": response.status_code,
        "duration_ms": round((time.perf_counter() - start) * 1000, 1),
    }))
    return response

with TestClient(logged_app) as client:
    first = client.post("/predict", json=requests_to_try["normal order"], headers={"X-Request-ID": "order-1234"})
    client.post("/predict", json=requests_to_try["hour 40"])
    print("header:", first.headers["X-Request-ID"])
```

Notice that the 422 was logged too: middleware wraps everything, including validation failures that never reach your endpoint. Logging the body itself is tempting but risky, since real request bodies often contain personal data; log identifiers and measurements instead.
:::

### Exercise 5: Review a broken Dockerfile

A teammate's Dockerfile builds, but it has at least five problems. Find them, say what each one causes, and write a fixed version. (There's no code to run; reason about it.)

```dockerfile
FROM python:latest
COPY . /app
WORKDIR /app
RUN pip install -r requirements-dev.txt
RUN python train.py
CMD uvicorn app.main:app
```

:::hint
Go line by line and ask: will this be the same next year? What happens to the cache when I edit one Python file? What ends up inside the image? Who does the server run as? Where does uvicorn listen by default?
:::

:::solution
1. **`python:latest`** changes whenever a new Python ships, so a rebuild can silently switch versions (and the full image is several times larger than `slim`). Pin `python:3.12-slim`.
2. **`COPY . /app` before installing** means any code edit invalidates the cache and reinstalls every library. It also copies `.venv`, `.git`, and any `.env` secrets unless a `.dockerignore` excludes them.
3. **`requirements-dev.txt`** ships pytest and httpx to production. Install `requirements.txt` only.
4. **`RUN python train.py`** trains a new, untested model at every build and needs the training data inside the image. Train outside, test, then copy the artifact in.
5. **`CMD uvicorn app.main:app`** listens on uvicorn's default `127.0.0.1`, which is unreachable from outside the container. It's also the *shell form*, which runs under `/bin/sh -c`; the JSON *exec form* runs uvicorn directly, so `docker stop` signals reach it and it can shut down cleanly.
6. **No `USER`**, so the server runs as root.

```dockerfile
FROM python:3.12-slim
ENV PYTHONDONTWRITEBYTECODE=1 PYTHONUNBUFFERED=1
WORKDIR /app
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt
COPY app/ app/
COPY models/ models/
RUN useradd --create-home appuser
USER appuser
EXPOSE 8000
CMD ["uvicorn", "app.main:app", "--host", "0.0.0.0", "--port", "8000"]
```
:::

```mermaid caption="Docker reuses cached layers until one's inputs change, then rebuilds from that layer down"

    A["FROM python:3.12-slim"] --> B["COPY requirements.txt"]
    B --> C["RUN pip install (slow)"]
```

What to watch for: official and pinned base images, ordering instructions for caching, `.dockerignore`, and running as a non-root user. Those four practices cover most of what's wrong in Exercise 5.

### Exercise 6: Measure the cost of loading per request

Someone suggests loading the model inside the handler "so it's always fresh". Measure what that costs. Build two apps that differ only in where `joblib.load` happens, time 20 requests to each, and do it for the small logistic regression and for a 150-tree random forest.

```text
logistic regression (     4 KB): load once   5.4 ms/request | load per request   8.1 ms/request
random forest       (11,674 KB): load once  13.3 ms/request | load per request  55.4 ms/request
```

Your numbers will differ, and both columns include the test client's own overhead per request; the gap between the columns is what matters.

:::hint
Write a function `build_apps(path)` that returns both apps: one loads the model into a variable before defining its endpoint, the other calls `joblib.load(path)` inside its endpoint. Send one warm-up request before starting the timer.
:::

:::solution
```python run
from sklearn.ensemble import RandomForestClassifier

forest = Pipeline([
    ("prep", ColumnTransformer([("hour", OneHotEncoder(handle_unknown="ignore"), ["hour"])],
                               remainder="passthrough")),
    ("clf", RandomForestClassifier(n_estimators=150, random_state=0)),
]).fit(orders[FEATURES], orders["late"])
joblib.dump(forest, "models/forest.joblib")

def build_apps(path):
    once, per_request = FastAPI(), FastAPI()
    loaded_model = joblib.load(path)

    @once.post("/predict")
    def predict_once(order: Order):
        return {"p": float(loaded_model.predict_proba(pd.DataFrame([order.model_dump()]))[0, 1])}

    @per_request.post("/predict")
    def predict_reloading(order: Order):
        fresh = joblib.load(path)
        return {"p": float(fresh.predict_proba(pd.DataFrame([order.model_dump()]))[0, 1])}

    return once, per_request

def ms_per_request(app, n=20):
    with TestClient(app) as client:
        client.post("/predict", json=requests_to_try["normal order"])     # warm-up
        start = time.perf_counter()
        for _ in range(n):
            client.post("/predict", json=requests_to_try["normal order"])
    return (time.perf_counter() - start) / n * 1000

for label, path in [("logistic regression", "models/model.joblib"), ("random forest", "models/forest.joblib")]:
    once, per_request = build_apps(path)
    size_kb = Path(path).stat().st_size / 1024
    print(f"{label:19} ({size_kb:>6,.0f} KB): load once {ms_per_request(once):5.1f} ms/request"
          f" | load per request {ms_per_request(per_request):5.1f} ms/request")
```

For a 4 KB model the reload costs a few milliseconds, which is how this mistake survives code review: it doesn't show up until the model grows. With the forest, most of each request is spent reading the file again. Big neural networks take seconds to load, so per-request loading would make the service unusable. "Always fresh" is the job of a deployment (a new container with the new model), not of every request.
:::

### Challenge: Refuse to start with the wrong scikit-learn

A model saved with one scikit-learn version and loaded with another may predict differently. Write `guarded_lifespan`, which reads `model.json` first and raises a `RuntimeError` naming both versions if `sklearn_version` doesn't match the installed one. Prove it by writing a `model.json` that claims `1.5.2`, show that the app refuses to start, then restore the real file.

```text
refused to start: model was saved with scikit-learn 1.5.2 but 1.8.0 is installed
after restoring model.json: 200
```

:::hint
Do the check before `yield`. An exception raised there makes `with TestClient(app)` raise too, the same way uvicorn would print `Application startup failed. Exiting.` and stop. Keep the original file's text in a variable so you can put it back.
:::

:::solution
```python run
@asynccontextmanager
async def guarded_lifespan(app):
    meta = json.loads(Path("models/model.json").read_text())
    if meta["sklearn_version"] != sklearn.__version__:
        raise RuntimeError(f"model was saved with scikit-learn {meta['sklearn_version']} "
                           f"but {sklearn.__version__} is installed")
    state["model"] = joblib.load("models/model.joblib")
    state["meta"] = meta
    yield
    state.clear()

guarded_app = FastAPI(lifespan=guarded_lifespan)

@guarded_app.post("/predict")
def guarded_predict(order: Order):
    return {"late_probability": round(float(score([order.model_dump()])[0]), 3)}

original = Path("models/model.json").read_text()
Path("models/model.json").write_text(json.dumps({**json.loads(original), "sklearn_version": "1.5.2"}))
try:
    with TestClient(guarded_app) as client:
        print("started (this should not happen)")
except RuntimeError as error:
    print("refused to start:", error)

Path("models/model.json").write_text(original)
with TestClient(guarded_app) as client:
    print("after restoring model.json:", client.post("/predict", json=requests_to_try["normal order"]).status_code)
```

Failing at startup turns a silent accuracy problem into a deploy that visibly doesn't come up, and the error message says exactly how to fix it: rebuild the image with the version the model was trained on, or retrain.
:::
