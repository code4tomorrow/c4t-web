## Step 1: Bring in the model and the tools

These three files are unchanged from Modules 1 and 2. Skip past them if you have them open. `harness/model.py`:

```python run title="harness/model.py"
"""The one interface every model in the harness speaks:

    reply = model.complete(messages, tools)

messages  a list of dicts such as {"role": "user", "content": "..."}
tools     a list of tool schemas (Module 2 builds these)
reply     {"role": "assistant", "content": "...",
           "tool_calls": [{"id": "...", "name": "...", "args": {...}}],
           "usage": {"input_tokens": 0, "output_tokens": 0}}
"""
import copy
import itertools
import json


def estimate_tokens(text):
    """About 4 characters per token in English. Rough, but fine for budgets."""
    return max(1, len(text) // 4)


_ids = itertools.count(1)


def say(text):
    """A scripted reply that is a final answer."""
    return {"role": "assistant", "content": text, "tool_calls": []}


def call(name, **args):
    """A scripted reply that asks for one tool call."""
    return {"role": "assistant", "content": "",
            "tool_calls": [{"id": f"call_{next(_ids)}", "name": name, "args": args}]}


class ScriptedModel:
    """Replies from a list you write in advance: offline, free, identical every run."""

    name = "scripted"

    def __init__(self, script):
        self.script = list(script)
        self.seen = []  # a copy of the messages it was shown, one list per call

    def complete(self, messages, tools=None):
        self.seen.append(copy.deepcopy(messages))
        if not self.script:
            raise RuntimeError("ScriptedModel ran out of replies")
        reply = self.script.pop(0)
        if callable(reply):  # a reply can be a function of what the model was shown
            reply = reply(messages)
        shown = json.dumps(messages) + json.dumps(tools or [])
        written = reply["content"] + json.dumps(reply["tool_calls"])
        usage = {"input_tokens": estimate_tokens(shown), "output_tokens": estimate_tokens(written)}
        return {**reply, "usage": usage}
```

`harness/tools.py`:

```python run title="harness/tools.py"
"""Tools the model may call: a schema it can read, a validator, and a dispatcher that never raises."""
from dataclasses import dataclass

TYPES = {"string": str, "integer": int, "number": (int, float),
         "boolean": bool, "array": list, "object": dict}


@dataclass
class Tool:
    name: str
    description: str
    params: dict      # {"path": {"type": "string", "description": "..."}}
    required: list
    fn: object        # the Python function to run

    def schema(self):
        return {"name": self.name, "description": self.description,
                "parameters": {"type": "object", "properties": self.params,
                               "required": self.required, "additionalProperties": False}}


def check_args(tool, args):
    """List what's wrong with args. An empty list means they are valid."""
    if not isinstance(args, dict):
        return ["arguments must be a JSON object"]
    problems = [f"missing required argument '{name}'" for name in tool.required if name not in args]
    for name, value in args.items():
        spec = tool.params.get(name)
        if spec is None:
            problems.append(f"unknown argument '{name}'")
            continue
        bool_as_number = isinstance(value, bool) and spec["type"] in ("integer", "number")
        if bool_as_number or not isinstance(value, TYPES[spec["type"]]):
            problems.append(f"'{name}' should be {spec['type']}, got {type(value).__name__}")
    return problems


def error(kind, message):
    return {"ok": False, "error": {"type": kind, "message": message}}


class ToolRegistry:
    def __init__(self):
        self.tools = {}

    def tool(self, name, description, params, required=None):
        """Decorator that registers a Python function as a tool."""
        def register(fn):
            needed = list(params) if required is None else required
            self.tools[name] = Tool(name, description, params, needed, fn)
            return fn
        return register

    def schemas(self):
        return [tool.schema() for tool in self.tools.values()]

    def dispatch(self, call):
        """Run one tool call. Every outcome, good or bad, comes back as a dict."""
        tool = self.tools.get(call["name"])
        if tool is None:
            return error("unknown_tool", f"no tool named '{call['name']}'; available: {', '.join(self.tools)}")
        problems = check_args(tool, call["args"])
        if problems:
            return error("bad_arguments", "; ".join(problems))
        try:
            return {"ok": True, "result": tool.fn(**call["args"])}
        except Exception as exc:  # the tool itself failed: report it, don't crash the run
            return error(type(exc).__name__, str(exc))
```

`harness/files.py`:

```python run title="harness/files.py"
"""File tools that only ever touch one workspace folder."""
from pathlib import Path


def add_file_tools(registry, root):
    root = Path(root).resolve()

    def inside(path):
        target = (root / path).resolve()
        if not target.is_relative_to(root):
            raise PermissionError(f"'{path}' is outside the workspace")
        return target

    @registry.tool("list_files", "List every file in the workspace.", {})
    def list_files():
        return sorted(p.relative_to(root).as_posix() for p in root.rglob("*") if p.is_file())

    @registry.tool("read_file", "Read a UTF-8 text file from the workspace.",
                   {"path": {"type": "string", "description": "Path relative to the workspace"}})
    def read_file(path):
        target = inside(path)
        if not target.is_file():
            raise FileNotFoundError(f"no file named '{path}'; call list_files to see what exists")
        return target.read_text(encoding="utf-8")

    @registry.tool("write_file", "Create or replace a UTF-8 text file in the workspace.",
                   {"path": {"type": "string"}, "content": {"type": "string"}})
    def write_file(path, content):
        target = inside(path)
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_text(content, encoding="utf-8")
        return f"wrote {len(content)} characters to {path}"
```

## Step 2: The tracer and the replay model

`harness/tracing.py` has three parts. `Tracer.event` opens the file in append mode, writes one line, and closes it, every time. That's slower than keeping the file open, but every event is on disk the moment it's logged, which is the point. `read_trace` stops at a half-written last line instead of crashing. `ReplayModel` turns the `model` events back into replies. The file needs no imports from the rest of the harness.

```python run title="harness/tracing.py"
"""Append-only JSONL traces: one event per line, written the moment it happens."""
import json
from pathlib import Path


class Tracer:
    def __init__(self, path, run_id):
        self.path, self.run_id = Path(path), run_id
        self.path.parent.mkdir(parents=True, exist_ok=True)

    def event(self, kind, step, **fields):
        record = {"run": self.run_id, "step": step, "type": kind, **fields}
        with self.path.open("a", encoding="utf-8") as file:
            file.write(json.dumps(record) + "\n")


def read_trace(path):
    """Load a trace. A half-written last line (the process died mid-write) is skipped."""
    events = []
    for line in Path(path).read_text(encoding="utf-8").splitlines():
        try:
            events.append(json.loads(line))
        except json.JSONDecodeError:
            break
    return events


class ReplayModel:
    """Plays back the model's side of a recorded run: same replies, no cost, no randomness."""

    name = "replay"

    def __init__(self, events):
        self.replies = [{"role": "assistant", "content": e["content"], "tool_calls": e["tool_calls"],
                         "usage": {"input_tokens": e["input_tokens"], "output_tokens": e["output_tokens"]}}
                        for e in events if e["type"] == "model"]

    def complete(self, messages, tools=None):
        if not self.replies:
            raise RuntimeError("the trace has no more model replies")
        return self.replies.pop(0)
```

## Step 3: A loop that records everything

This replaces `harness/loop.py` from Module 2. The logic is the same; what's new is a `tracer` argument and a `log` call at each point where something happens. When no tracer is passed, `log` is a function that does nothing, so old callers and tests keep working.

```python run title="harness/loop.py"
"""The runtime loop, now writing a trace event for everything it observes and does."""
import json
import time
from dataclasses import dataclass


@dataclass
class RunResult:
    status: str      # "done", "max_steps", or "model_error"
    answer: str
    steps: int
    messages: list


SYSTEM = ("You are a careful coding assistant. Use the tools to inspect and change files in "
          "the workspace. When the task is finished, reply with a short summary and no tool calls.")


def _ms(started):
    return round((time.perf_counter() - started) * 1000, 2)


def run_agent(model, registry, task, system=SYSTEM, max_steps=8, verify=None, tracer=None):
    log = tracer.event if tracer else (lambda kind, step, **fields: None)
    log("start", 0, task=task, model=getattr(model, "name", "?"),
        tools=list(registry.tools), max_steps=max_steps)
    messages = [{"role": "system", "content": system}, {"role": "user", "content": task}]
    result = RunResult("max_steps", "", max_steps, messages)
    for step in range(1, max_steps + 1):
        started = time.perf_counter()
        try:
            reply = model.complete(messages, registry.schemas())
        except Exception as exc:
            result = RunResult("model_error", f"{type(exc).__name__}: {exc}", step, messages)
            break
        log("model", step, ms=_ms(started), content=reply["content"],
            tool_calls=reply["tool_calls"], **reply["usage"])
        messages.append({"role": "assistant", "content": reply["content"],
                         "tool_calls": reply["tool_calls"]})
        if reply["tool_calls"]:
            for call in reply["tool_calls"]:
                started = time.perf_counter()
                outcome = registry.dispatch(call)
                log("tool", step, ms=_ms(started), call_id=call["id"], tool=call["name"],
                    args=call["args"], **outcome)
                messages.append({"role": "tool", "tool_call_id": call["id"],
                                 "name": call["name"], "content": json.dumps(outcome)})
            continue
        problem = verify() if verify else None
        if verify:
            log("verify", step, ok=problem is None, problem=problem)
        if problem is None:
            result = RunResult("done", reply["content"], step, messages)
            break
        messages.append({"role": "user", "content": f"Verification failed: {problem}"})
    log("end", result.steps, status=result.status, answer=result.answer)
    return result
```

Three details. Latency is measured with `time.perf_counter`, which is meant for timing, unlike `time.time`. The tool event uses `**outcome`, so it records `ok` and either `result` or `error` exactly as the model saw them. And the loop now ends with `break` instead of several `return`s, so the `end` event is written on every path, including `model_error` and `max_steps`.

## Step 4: Record six runs

Six tasks, each with a scripted agent standing in for a real model's run. They're chosen to cover the cases you'll want to measure: clean successes, a lucky guess, a tool error the agent recovers from, and an agent that never stops. `fresh_workspace` gives each run its own starting files.

```python run
import shutil


def fresh_workspace(files):
    shutil.rmtree("workspace", ignore_errors=True)
    Path("workspace").mkdir()
    for name, text in files.items():
        Path("workspace", name).write_text(text, encoding="utf-8")
    registry = ToolRegistry()
    add_file_tools(registry, "workspace")
    return registry


NOTES = {"notes.txt": "Project: weather bot\nDeadline: June 3\nOwner: Priya\n"}
TASKS = {
    "deadline": ("When is the project deadline?", NOTES),
    "fix-add": ("Fix calc.py so add(2, 3) returns 5.", {"calc.py": "def add(a, b):\n    return a - b\n"}),
    "owner": ("Who owns the project?", NOTES),
    "summary": ("Write a one-line summary of notes.txt to summary.txt.", NOTES),
    "missing": ("What does todo.txt say?", NOTES),
    "tidy": ("Tidy up the workspace.", NOTES),
}
AGENTS = {   # one scripted agent per task, standing in for a real model's runs
    "deadline": lambda: [call("list_files"), call("read_file", path="notes.txt"),
                         say("The deadline is June 3.")],
    "fix-add": lambda: [call("read_file", path="calc.py"),
                        call("write_file", path="calc.py", content="def add(a, b):\n    return a + b\n"),
                        say("Fixed: add() now returns a + b.")],
    "owner": lambda: [say("Priya owns the project.")],
    "summary": lambda: [call("read_file", path="notes.txt"),
                        call("write_file", path="summary.txt", content="Weather bot, due June 3, owned by Priya.\n"),
                        say("Wrote summary.txt.")],
    "missing": lambda: [call("read_file", path="todo.txt"), call("list_files"),
                        say("There is no todo.txt in the workspace. The only file is notes.txt.")],
    "tidy": lambda: [call("list_files") for _ in range(10)],
}

shutil.rmtree("traces", ignore_errors=True)

    tracer = Tracer(f"traces/{task_id}.jsonl", run_id=task_id)

## Step 5: Read the traces

Traces are data, so questions about runs become a few lines of code. `summarize` turns one trace into a row:

```python run
def summarize(events):
    models = [e for e in events if e["type"] == "model"]
    tools = [e for e in events if e["type"] == "tool"]
    end = events[-1]
    return {"status": end["status"], "steps": end["step"], "tool_calls": len(tools),
            "tool_errors": sum(not e["ok"] for e in tools),
            "tokens_in": sum(e["input_tokens"] for e in models),
            "tokens_out": sum(e["output_tokens"] for e in models),
            "ms": round(sum(e["ms"] for e in models + tools), 1)}


traces = {task_id: read_trace(f"traces/{task_id}.jsonl") for task_id in TASKS}
summaries = {task_id: summarize(events) for task_id, events in traces.items()}
print(f"{'task':<9} {'status':<10} {'steps':>5} {'calls':>5} {'errors':>6} {'tok in':>7} {'tok out':>7}")
for task_id, s in summaries.items():
    print(f"{task_id:<9} {s['status']:<10} {s['steps']:>5} {s['tool_calls']:>5} {s['tool_errors']:>6} "

## Step 6: Replay a run

Replay the `deadline` run twice: once with the same files, once after the deadline in `notes.txt` has moved.

```python run
def first_divergence(old_events, new_events):
    old = [e for e in old_events if e["type"] == "tool"]
    new = [e for e in new_events if e["type"] == "tool"]
    for a, b in zip(old, new):
        if (a["tool"], a["args"], a.get("result"), a.get("error")) != (b["tool"], b["args"], b.get("result"), b.get("error")):
            return a["step"], a["tool"], a.get("result", a.get("error")), b.get("result", b.get("error"))
    return None


recorded = traces["deadline"]
registry = fresh_workspace(NOTES)
replay = run_agent(ReplayModel(recorded), registry, recorded[0]["task"], max_steps=6,
                   tracer=Tracer("traces/deadline-replay.jsonl", run_id="deadline-replay"))
print(f"same world:    answer {replay.answer!r}, first divergence:",
      first_divergence(recorded, read_trace("traces/deadline-replay.jsonl")))

registry = fresh_workspace({"notes.txt": "Project: weather bot\nDeadline: June 10\nOwner: Priya\n"})
replay = run_agent(ReplayModel(recorded), registry, recorded[0]["task"], max_steps=6,
                   tracer=Tracer("traces/deadline-replay2.jsonl", run_id="deadline-replay2"))
step, tool, before, after = first_divergence(recorded, read_trace("traces/deadline-replay2.jsonl"))
print(f"changed world: answer {replay.answer!r}, first divergence at step {step} ({tool})")
print("  recorded:", repr(before))
print("  replayed:", repr(after))
```

In the same world, the replay matched the recording event for event. In the changed world, it diverged at step 2, where `read_file` returned the new text, yet the replayed answer still said June 3, because a recording can't react. That's replay's job: holding the model fixed so the difference points at what changed around it.

## Step 7: Freeze the eval set

`harness/evals.py` turns traces into cases and grades runs against them. `check` runs the deterministic checks; its `says` helper matches whole words, so "June 3" doesn't match "June 30". `judge` asks a model for a verdict and treats anything except a clean JSON `true` or `false` as a failed grade. `grade` combines them. The file needs no imports from the rest of the harness.

```python run title="harness/evals.py"
"""Eval cases built from traces, deterministic checks, and a model-based judge."""
import hashlib
import json
import re
from pathlib import Path


def case_from_trace(events, case_id, files, expect):
    start = events[0]
    return {"id": case_id, "task": start["task"], "files": files,
            "expect": expect, "source_run": start["run"]}


def save_cases(cases, path):
    """Write cases in a stable order and return the file's version."""
    lines = [json.dumps(case, sort_keys=True) for case in sorted(cases, key=lambda c: c["id"])]
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    Path(path).write_text("\n".join(lines) + "\n", encoding="utf-8")
    return dataset_version(path)


def dataset_version(path):
    """A fingerprint of the exact bytes of the case file."""
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()[:12]


def load_cases(path):
    return [json.loads(line) for line in Path(path).read_text(encoding="utf-8").splitlines() if line]


def says(text, phrase):
    """True if phrase appears as whole words, so "June 3" doesn't match "June 30"."""
    return re.search(rf"\b{re.escape(phrase)}\b", text, re.IGNORECASE) is not None


def check(case, result, events, workspace):
    """Deterministic checks: the same run always gets the same answers."""
    expect, checks = case["expect"], {"finished": result.status == "done"}
    if "answer_contains" in expect:
        checks["answer"] = all(says(result.answer, phrase) for phrase in expect["answer_contains"])
    if "must_call" in expect:
        called = {e["tool"] for e in events if e["type"] == "tool"}
        checks["tools"] = set(expect["must_call"]) <= called
    if "max_steps" in expect:
        checks["steps"] = result.steps <= expect["max_steps"]
    for name, phrases in expect.get("file_contains", {}).items():
        path = Path(workspace) / name
        checks[f"file:{name}"] = path.is_file() and all(p in path.read_text(encoding="utf-8") for p in phrases)
    return checks


JUDGE_PROMPT = """You are grading an assistant's answer. Judge it only against the rubric.
Rubric: {rubric}
Task: {task}
Answer: {answer}
Reply with JSON only, exactly like {{"pass": true, "reason": "one sentence"}}."""


def judge(model, rubric, task, answer):
    """Ask a model for a verdict. Anything but a clean verdict is a failed grade, never a pass."""
    reply = model.complete([{"role": "user", "content": JUDGE_PROMPT.format(rubric=rubric, task=task, answer=answer)}])
    try:
        verdict = json.loads(reply["content"])
    except json.JSONDecodeError:
        return {"pass": False, "reason": "judge did not reply with JSON", "judge_error": True}
    if not isinstance(verdict, dict) or not isinstance(verdict.get("pass"), bool):
        return {"pass": False, "reason": "judge reply had no true/false pass", "judge_error": True}
    return {"pass": verdict["pass"], "reason": str(verdict.get("reason", "")), "judge_error": False}


def grade(case, result, events, workspace, judge_model=None):
    checks, reason = check(case, result, events, workspace), ""
    if "rubric" in case["expect"]:
        if judge_model is None:
            checks["judge"], reason = False, "this case needs a judge model"
        else:
            verdict = judge(judge_model, case["expect"]["rubric"], case["task"], result.answer)
            checks["judge"], reason = verdict["pass"], verdict["reason"]
    return {"case": case["id"], "pass": all(checks.values()), "checks": checks, "judge_reason": reason}
```

Now write down what a correct run must do for each task, and freeze the cases. The `missing` task's success condition is about meaning (don't make up the contents of a file that doesn't exist), so it gets a rubric for the judge instead of a phrase to match:

    "owner": {"answer_contains": ["Priya"], "must_call": ["read_file"]},
    "summary": {"file_contains": {"summary.txt": ["June 3", "Priya"]}, "max_steps": 4},

cases = [case_from_trace(traces[task_id], task_id, TASKS[task_id][1], EXPECT[task_id]) for task_id in TASKS]
version = save_cases(cases, "evals/tasks-v1.jsonl")
print("evals/tasks-v1.jsonl version", version)
print(Path("evals/tasks-v1.jsonl").read_text().splitlines()[0])
```

`sort_keys=True` and sorting by id mean the same cases always produce the same bytes, so the version only changes when the content does. Each case also records `source_run`, the trace it came from, so you can always look up the run that inspired it.

## Step 8: The judge, and what it does with bad replies

Before trusting the judge, check how it handles a reply that isn't a verdict. A scripted judge gives one good reply and two bad ones:

```python run
rubric = EXPECT["missing"]["rubric"]
answer = results["missing"].answer
replies = ScriptedModel([
    say('{"pass": true, "reason": "Says todo.txt does not exist and names the file that does."}'),
    say("Looks good to me! I'd give it a pass."),
    say('{"verdict": "pass"}'),
])
for _ in range(3):
    print(judge(replies, rubric, TASKS["missing"][0], answer))
```

The chatty reply and the reply with the wrong key both became failed grades marked `judge_error`. Counting them as passes would let a broken judge inflate your score; counting them separately lets you see when the judge itself needs attention.

## The finished program

`run_evals.py` loads the frozen file, runs each case in a fresh workspace with its own trace, grades it, and prints the scorecard with the version. On disk it imports `fresh_workspace` and `AGENTS` from wherever you keep them (an `agents.py` next to it works), plus `harness.chat_api`, `harness.evals`, `harness.loop`, `harness.model`, and `harness.tracing`.

```python run title="run_evals.py"
import sys


def run_evals(path, agent_for, judge_model):
    version = dataset_version(path)
    grades = []
    for case in load_cases(path):
        registry = fresh_workspace(case["files"])
        tracer = Tracer(f"traces/eval-{version}-{case['id']}.jsonl", run_id=f"eval-{case['id']}")
        result = run_agent(agent_for(case), registry, case["task"], max_steps=6, tracer=tracer)
        grades.append(grade(case, result, read_trace(tracer.path), "workspace", judge_model))
    return version, grades


def main(live=False):
    if live:
        agent_for, judge_model = (lambda case: ChatAPIModel()), ChatAPIModel()
    else:
        agent_for = lambda case: ScriptedModel(AGENTS[case["id"]]())
        judge_model = ScriptedModel([say('{"pass": true, "reason": "Says todo.txt does not exist '
                                         'instead of inventing its contents."}')])
    version, grades = run_evals("evals/tasks-v1.jsonl", agent_for, judge_model)
    for g in grades:
        failed = [name for name, ok in g["checks"].items() if not ok]
        note = f'  judge: {g["judge_reason"]}' if g["judge_reason"] else ""
        print(f"{g['case']:<9} {'PASS' if g['pass'] else 'FAIL'}  {'failed: ' + ', '.join(failed) if failed else ''}{note}")
    passed = sum(g["pass"] for g in grades)
    print(f"\nscore {passed}/{len(grades)} on tasks-v1.jsonl @ {version}")


main(live="--live" in sys.argv)
```

Four of six. `tidy` failed because it never finished. `owner` failed on `tools`: it answered "Priya" correctly without reading anything, which is a guess, and the `must_call` check is what separates a guess from an answer. A score that only checked the final answer would have passed it. Every eval run also left traces named with the dataset version, so any line of the scorecard can be traced back to the exact run that produced it.

With `--live`, every case runs against your real model and the judge is a real model too. Expect the score to change from run to run; Module 5 covers how to tell a real change from noise.

:::quiz
Q: Why does `Tracer.event` open and close the file for every event instead of keeping it open?
- [x] So each event is on disk as soon as it's logged, and a crash loses at most the event being written
  Buffered writes to an open file can sit in memory, and a crash would lose them along with the most important part of the trace.
- [ ] Because JSON Lines files can't stay open
  Any file can stay open; the choice is about durability.
- [ ] To make tracing faster
  It's a little slower. The trade is speed for a trace that survives crashes.

Q: The `owner` run answered correctly in one step. Why does it fail the eval?
- [ ] Its answer doesn't contain "Priya"
  It does; the `answer` check passed.
- [x] It never called `read_file`, so the correct answer was a guess, and the `must_call` check fails it
  Checking the path as well as the answer separates luck from work.
- [ ] One step is below the minimum
  There's no minimum; `max_steps` is only an upper limit.

Q: A judge replies "Looks good to me! I'd give it a pass." What does `grade` record?
- [ ] A pass, because the reply clearly approves
  Reading approval into free text is exactly the guessing a strict judge avoids.
- [x] A failed `judge` check marked as a judge error, so it neither inflates the score nor hides the problem
  Parse failures are tracked separately so you can fix the judge rather than trust it.
:::

:::takeaways
- `Tracer` appends one JSON line per event, and the traced loop logs start, every model and tool call, verification, and end on every path.
- Traces are data: summaries, token counts, and charts are a few lines of code.
- Steps and tokens measure cost, not correctness; a cheap run can still be a wrong or lucky one.
- `ReplayModel` reproduces a run for free, and the first divergence shows what changed in the world.
- The eval set is frozen in `tasks-v1.jsonl` with a version hash, and every score is printed with it.
- The judge only passes on a clean `true`; anything else is a failed grade marked as a judge error.
:::
